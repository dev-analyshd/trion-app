    24→fn sha3_256(data: &[u8]) -> [u8; 32] {
    25→    Sha3_256::digest(data).into()
    26→}

    28→fn not_bytes(b: &[u8; 32]) -> [u8; 32] {
    29→    let mut out = [0u8; 32];
    30→    for i in 0..32 { out[i] = !b[i]; }
    31→    out
    32→}
    33→
    34→fn xor_bytes(a: &[u8; 32], b: &[u8; 32]) -> [u8; 32] {
    35→    let mut out = [0u8; 32];
    36→    for i in 0..32 { out[i] = a[i] ^ b[i]; }
    37→    out
    38→}

    44→// ── Component 2: Complementary Strand Verification ───────────────────────────

    46→/// Dual-strand DNA hash — whitepaper L0.1.
    47→/// sense     = SHA3-256(payload || 0x00)
    48→/// antisense = SHA3-256(payload || 0xFF) XOR NOT(sense)
    49→/// Invariant: sense XOR antisense == NOT(SHA3-256(payload || 0xFF))
    50→#[derive(Clone, Debug)]
    51→pub struct DualStrand {
    52→    pub sense: [u8; 32],
    53→    pub antisense: [u8; 32],
    54→}
    55→
    56→impl DualStrand {
    57→    pub fn compute(payload: &[u8]) -> Self {
    58→        let mut p0 = payload.to_vec();  p0.push(0x00);
    59→        let mut pff = payload.to_vec(); pff.push(0xFF);
    60→        let sense: [u8; 32] = sha3_256(&p0);
    61→        let sha3ff: [u8; 32] = sha3_256(&pff);
    62→        let antisense = xor_bytes(&sha3ff, &not_bytes(&sense));
    63→        Self { sense, antisense }
    64→    }
    65→
    66→    /// Verify complement invariant WITHOUT needing the original payload.
    67→    /// antisense == SHA3FF XOR NOT(sense)
    68→    /// => sense XOR antisense == NOT(SHA3FF)
    69→    /// We verify structural self-consistency: antisense is deterministically
    70→    /// derivable from sense and sha3ff. We check the length and non-zero structure.
    71→    /// Full verification requires the original payload (see verify_with_payload).
    72→    pub fn verify_structural(&self) -> bool {
    73→        // Structural: lengths OK and not all-zero (degenerate)
    74→        self.sense != [0u8; 32] && self.antisense != [0u8; 32]
    75→    }
    76→
    77→    /// Full cryptographic verification — requires the original payload.
    78→    /// Recomputes both strands and checks equality.
    79→    pub fn verify_with_payload(&self, payload: &[u8]) -> bool {
    80→        let recomputed = Self::compute(payload);
    81→        recomputed.sense == self.sense && recomputed.antisense == self.antisense
    82→    }
    83→
    84→    /// Verify the XOR complement invariant:
    85→    /// sense XOR antisense == NOT(SHA3-256(payload || 0xFF))
    86→    pub fn verify_xor_invariant(&self, payload: &[u8]) -> bool {
    87→        let mut pff = payload.to_vec(); pff.push(0xFF);
    88→        let sha3ff: [u8; 32] = sha3_256(&pff);
    89→        let expected_xor = not_bytes(&sha3ff); // NOT(SHA3FF)
    90→        let actual_xor = xor_bytes(&self.sense, &self.antisense);
    91→        actual_xor == expected_xor
    92→    }
    93→
    94→    pub fn sense_hex(&self) -> String { hex::encode(self.sense) }
    95→    pub fn antisense_hex(&self) -> String { hex::encode(self.antisense) }
    96→}

   510→    pub integrity_checks: u64,
   511→    pub last_check: u64,
   512→}
   513→
   514→impl MitochondrialCore {
   515→    pub fn new(protocol_version: u32, chain_count: u32) -> Self {
   516→        let genesis = now_secs();
   517→        let mut payload = Vec::with_capacity(16);
   518→        payload.extend_from_slice(&protocol_version.to_be_bytes());
   519→        payload.extend_from_slice(&chain_count.to_be_bytes());
   520→        payload.extend_from_slice(&genesis.to_be_bytes());
   521→        payload.extend_from_slice(b"TRION_MITO_CORE_v3");
   522→        let core_dna = DualStrand::compute(&payload);
   523→        Self {
   524→            core_dna,
   525→            protocol_version,
   526→            chain_count,
   527→            genesis_timestamp: genesis,
   528→            integrity_checks: 0,
   529→            last_check: genesis,
   530→        }
   531→    }
   532→
   533→    /// Verify the mitochondrial core integrity (independent of GK).
   534→    pub fn verify_integrity(&mut self) -> bool {
   535→        self.integrity_checks += 1;
   536→        self.last_check = now_secs();
   537→        self.core_dna.verify_structural()
   538→    }
   539→
   540→    /// Update core when chain count or protocol version changes.
   541→    pub fn update(&mut self, new_chain_count: u32) {
   542→        self.chain_count = new_chain_count;
   543→        let mut payload = Vec::with_capacity(24);
   544→        payload.extend_from_slice(&self.protocol_version.to_be_bytes());
   545→        payload.extend_from_slice(&new_chain_count.to_be_bytes());
   546→        payload.extend_from_slice(&now_secs().to_be_bytes());
   547→        payload.extend_from_slice(&self.core_dna.sense); // chained to previous
   548→        payload.extend_from_slice(b"TRION_MITO_CORE_v3");
   549→        self.core_dna = DualStrand::compute(&payload);
   550→    }
   551→
   552→    pub fn integrity_score(&self) -> f64 {
   553→        if self.core_dna.verify_structural() { 1.0 } else { 0.0 }
   554→    }
   555→}

   559→/// Post-Quantum Cryptography score (whitepaper L4.5).
   560→/// CRYSTALS-Kyber + CRYSTALS-Dilithium + SPHINCS+
   561→/// In production: use actual PQC libraries. Here: structural representation.
   562→#[derive(Clone, Debug)]
   563→pub struct PQCScore {
   564→    pub kyber_active: bool,    // CRYSTALS-Kyber (KEM)
   565→    pub dilithium_active: bool, // CRYSTALS-Dilithium (signature)
   566→    pub sphincs_active: bool,  // SPHINCS+ (hash-based signature)
   567→    pub score: f64,
   568→}
   569→
   570→impl PQCScore {
   571→    pub fn new(kyber: bool, dilithium: bool, sphincs: bool) -> Self {
   572→        let score = (kyber as u8 as f64 + dilithium as u8 as f64 + sphincs as u8 as f64) / 3.0;
   573→        Self { kyber_active: kyber, dilithium_active: dilithium, sphincs_active: sphincs, score }
   574→    }
   575→    pub fn all_active() -> Self { Self::new(true, true, true) }
   576→}

   589→    pub fn new(sha3: bool, aes256: bool, zk: bool) -> Self {

   596→/// Full security score: SEC(t) = LSS(t) · PQC(t) · CC(t)
   597→pub fn compute_sec(
   598→    gk: &GenomicKey,
   599→    epigenetic: &EpigeneticLayer,
   600→    mito: &MitochondrialCore,
   601→    crispr_library_size: usize,
   602→    pqc: &PQCScore,
   603→    cc: &ClassicalCryptoScore,
   604→) -> SecurityScore {
   605→    // LSS: Living Security Score
   606→    // Based on: GK generation (causal depth), epigenetic health, mitochondrial integrity
   607→    let gk_depth = (gk.generation as f64).ln_1p() / (100.0_f64).ln_1p(); // normalize to ~[0,1]
   608→    let gk_depth = gk_depth.min(1.0);
   609→    let epi_health = 1.0 - epigenetic.threat_level * 0.5;
   610→    let mito_integrity = mito.integrity_score();
   611→    let crispr_coverage = (crispr_library_size as f64 / 8.0).min(1.0); // 8 = minimum seeded
   612→    let lss = (gk_depth * 0.40 + epi_health * 0.25 + mito_integrity * 0.20 + crispr_coverage * 0.15)
   613→        .clamp(0.0, 1.0);
   614→
   615→    // P(break LSS) is monotonically decreasing: approximated as e^(-generation)
   616→    let p_break_lss = (-(gk.generation as f64) * 0.01).exp().min(0.999);
   617→
   618→    let sec = lss * pqc.score * cc.score;
   619→
   620→    SecurityScore {
   621→        lss,
   622→        pqc: pqc.score,
   623→        cc: cc.score,
   624→        sec,
   625→        p_break_lss,
   626→        gk_generation: gk.generation,
   627→        epigenetic_state: epigenetic.state_name().to_string(),
   628→        mito_integrity: mito_integrity,
   629→        crispr_library_size,
   630→    }
   631→}

   633→#[derive(Clone, Debug)]
   634→pub struct SecurityScore {
   635→    pub lss: f64,
   636→    pub pqc: f64,
   637→    pub cc: f64,
   638→    pub sec: f64,
   639→    pub p_break_lss: f64,
   640→    pub gk_generation: u64,
   641→    pub epigenetic_state: String,
   642→    pub mito_integrity: f64,
   643→    pub crispr_library_size: usize,
   644→}

   646→// ── Bootstrap Protocol (L4.7) ─────────────────────────────────────────────────
   647→
   648→/// SEC_boot(t) = bootstrap_weight(t) · SEC_classical + (1 - bootstrap_weight(t)) · SEC_living
   649→/// bootstrap_weight(t) = e^(-λ_boot · D(t))
   650→/// At D(t) = D_minimum (~6 months): bootstrap_weight ≈ 0 → Living Security fully active
   651→pub fn bootstrap_weight(akashic_depth: u64) -> f64 {
   652→    let lambda_boot = 0.0001; // tuned so D=10000 blocks → weight ≈ 0.37, D=50000 → weight ≈ 0.007
   653→    (-lambda_boot * akashic_depth as f64).exp()
   654→}
   655→
   656→pub fn sec_bootstrap(
   657→    akashic_depth: u64,
   658→    sec_classical: f64,  // multi-sig 7-of-12 + rate limit + human oversight
   659→    sec_living: f64,     // SEC(t) from compute_sec
   660→) -> f64 {
   661→    let w = bootstrap_weight(akashic_depth);
   662→    w * sec_classical + (1.0 - w) * sec_living
   663→}
