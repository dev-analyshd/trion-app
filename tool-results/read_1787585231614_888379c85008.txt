     7→     7→ *
     8→     8→ * This contract is the on-chain core of the TRION × 0G execution intelligence
     9→     9→ * system. AI agents must query this gate before executing any DeFi trade.
    10→    10→ * If behavioral entropy is elevated, execution is blocked on-chain — not just
    11→    11→ * warned about, but made IMPOSSIBLE until conditions normalize.
    12→    12→ *
    13→    13→ * Architecture:
    14→    14→ *  - TRION behavioral oracle scores entity BEO DNA off-chain (FAISS, 531K vectors)

    18→    18→ *  - 0G Storage holds the full behavioral memory (BEO vector index)

    26→    26→ *  - Fail-closed — uninitialized entities are BLOCKED, not allowed
    27→    27→ *  - OwnershipTransferred event emitted on acceptance
    28→    28→ *
    29→    29→ *  ─────────────────────────────────────────────────────────────────
    30→    30→ *  AUDIT-3 Gap G3 Fix — AWA Enforcement In Signal Publication Path

    37→    37→ *  the top of `publishSignal()`.
    38→    38→ *
    39→    39→ *  The AWA state (HHI, gratitude score, public-good %, validator count)
    40→    40→ *  is updated by the Akashic oracle via `updateAWAState()` and checked
    41→    41→ *  atomically on every signal publication.

   230→   230→    // ── AWA Enforcement (AUDIT-3 Gap G3) ─────────────────────────────────────
   231→   231→    /**
   232→   232→     * @notice Returns TRUE iff all four Archetypal Weighted Average (AWA)
   233→   233→     *         governance conditions are satisfied. WP2 §17 requires this to
   234→   234→     *         be TRUE before any signal can be published; if FALSE, signal
   235→   235→     *         emission is FROZEN and cannot be overridden by any single
   236→   236→     *         entity (including the owner).
   237→   237→     *
   238→   238→     *         Conditions:
   239→   239→     *           1. Quorum: configured `quorumRequired` ≥ ⌈2/3 · validatorCount⌉
   240→   240→     *           2. HHI:    `currentHHI` < 4000 (critical threshold)
   241→   241→     *           3. Gratitude: `gratitudeScore` ≥ 1
   242→   242→     *           4. Public Good: `publicGoodBps` ≥ 1500 (15% charter)
   243→   243→     *
   244→   244→     * @return enforced TRUE iff all four conditions hold.
   245→   245→     */
   246→   246→    function awaEnforced() public view returns (bool enforced) {
   247→   247→        // (1) Quorum: the configured quorum must itself be a 2/3 supermajority
   248→   248→        //     of the validator set. Without this, a small cabal could publish
   249→   249→        //     signals even with an under-decentralized validator mesh.
   250→   250→        if (validatorCount == 0) return false;
   251→   251→        uint256 requiredQuorum = (validatorCount * AWA_QUORUM_NUM + (AWA_QUORUM_DEN - 1)) / AWA_QUORUM_DEN;
   252→   252→        if (quorumRequired < requiredQuorum) return false;
   253→   253→
   254→   254→        // (2) HHI: must be below the critical 4000 threshold.
   255→   255→        if (currentHHI >= AWA_HHI_MAX) return false;
   256→   256→
   257→   257→        // (3) Gratitude: ≥ 1 (Love Protocol multiplicative ethics floor).
   258→   258→        if (gratitudeScore < AWA_GRATITUDE_MIN) return false;
   259→   259→
   260→   260→        // (4) Public Good: ≥ 15% charter allocation.
   261→   261→        if (publicGoodBps < AWA_PUBLIC_GOOD_MIN_BPS) return false;
   262→   262→
   263→   263→        return true;
   264→   264→    }
   265→   265→
   266→   266→    /**
   267→   267→     * @notice Returns a human-readable diagnostic of which AWA condition(s)
   268→   268→     *         are currently failing. Used by the dashboard / monitoring.
   269→   269→     *         Bitmask: bit 0 = quorum, bit 1 = HHI, bit 2 = gratitude, bit 3 = publicGood.
   270→   270→     *         0 means all conditions satisfied.
   271→   271→     */
   272→   272→    function awaFailureMask() external view returns (uint256 mask) {
   273→   273→        if (validatorCount == 0) {
   274→   274→            mask |= 1;
   275→   275→        } else {
   276→   276→            uint256 requiredQuorum = (validatorCount * AWA_QUORUM_NUM + (AWA_QUORUM_DEN - 1)) / AWA_QUORUM_DEN;
   277→   277→            if (quorumRequired < requiredQuorum) mask |= 1;
   278→   278→        }
   279→   279→        if (currentHHI >= AWA_HHI_MAX)               mask |= 2;
   280→   280→        if (gratitudeScore < AWA_GRATITUDE_MIN)      mask |= 4;
   281→   281→        if (publicGoodBps < AWA_PUBLIC_GOOD_MIN_BPS) mask |= 8;
   282→   282→    }

   301→   301→    // ── Core: Publish Behavioral Signal (with quorum + AWA enforcement) ─────
   302→   302→    /**
   303→   303→     * @notice Validators publish a TRION behavioral signal for an entity.
   304→   304→     *         Requires `quorumRequired` valid EIP-191 signatures over the
   305→   305→     *         message hash keccak256(abi.encodePacked(
   306→   306→     *           block.chainid, address(this), entityId, packedData
   307→   307→     *         )) from distinct registered validators.
   308→   308→     *
   309→   309→     *         AUDIT-3 Gap G3: also requires `awaEnforced()` to be TRUE.
   310→   310→     *         If any AWA condition fails, signal emission is FROZEN.
   311→   311→     *
   312→   312→     * @param entityId    keccak256 BEO entity identifier
   313→   313→     * @param packedData  Packed behavioral metrics (phi_t, theta, drop_pct, status)
   314→   314→     * @param beoHash     keccak256 of entity's behavioral DNA (content-addressed on 0G Storage)
   315→   315→     * @param daProofHash 0G DA content hash — proof this anomaly is permanently sealed
   316→   316→     * @param storageRoot 0G Storage merkle root for this entity's behavioral memory
   317→   317→     * @param signatures  Ordered array of EIP-191 signatures from distinct validators
   318→   318→     */
   319→   319→    function publishSignal(
   320→   320→        bytes32 entityId,
   321→   321→        uint256 packedData,
   322→   322→        bytes32 beoHash,
   323→   323→        bytes32 daProofHash,
   324→   324→        string calldata storageRoot,
   325→   325→        bytes[] calldata signatures
   326→   326→    ) external onlyValidator whenNotPaused {
   327→   327→        // ── AWA enforcement (AUDIT-3 Gap G3) ────────────────────────────────
   328→   328→        // WP2 §17: "AWA_enforced = FALSE → signal emission FROZEN. Cannot be
   329→   329→        // overridden by any single entity." Check this BEFORE the quorum
   330→   330→        // signature loop so a frozen state short-circuits cheaply.
   331→   331→        require(awaEnforced(), "TRION: AWA not enforced - signal emission frozen");
   332→   332→
   333→   333→        // ── Quorum enforcement ────────────────────────────────────────────────
   334→   334→        require(
   335→   335→            signatures.length >= quorumRequired,
   336→   336→            "TRION: Insufficient signatures for quorum"
   337→   337→        );
   338→   338→
   339→   339→        bytes32 msgHash = keccak256(abi.encodePacked(
   340→   340→            block.chainid, address(this), entityId, packedData
   341→   341→        ));
   342→   342→        bytes32 ethSignedHash = keccak256(abi.encodePacked(
   343→   343→            "\x19Ethereum Signed Message:\n32", msgHash
   344→   344→        ));
   345→   345→
   346→   346→        address[] memory seen = new address[](signatures.length);
   347→   347→        uint256 validCount = 0;
   348→   348→
   349→   349→        for (uint256 i = 0; i < signatures.length; i++) {
   350→   350→            address recovered = _recoverSigner(ethSignedHash, signatures[i]);
   351→   351→            if (!isValidator[recovered]) continue;
   352→   352→
   353→   353→            // Prevent duplicate signers
   354→   354→            bool duplicate = false;
   355→   355→            for (uint256 j = 0; j < validCount; j++) {
   356→   356→                if (seen[j] == recovered) { duplicate = true; break; }
   357→   357→            }
   358→   358→            if (duplicate) continue;
   359→   359→
   360→   360→            seen[validCount] = recovered;
   361→   361→            validCount++;
   362→   362→            if (validCount >= quorumRequired) break;
   363→   363→        }
   364→   364→
   365→   365→        require(validCount >= quorumRequired, "TRION: Quorum not met by distinct validators");
   366→   366→
   367→   367→        // ── Unpack and validate ───────────────────────────────────────────────
   368→   368→        uint8  status  = uint8(packedData & 0xFF);
   369→   369→        uint32 phi_t   = uint32((packedData >> 8)  & 0xFFFFFFFF);
   370→   370→        uint32 theta   = uint32((packedData >> 40) & 0xFFFFFFFF);
   371→   371→        uint32 dropPct = uint32((packedData >> 72) & 0xFFFFFFFF);
   372→   372→
   373→   373→        require(status >= 1 && status <= 4, "TRION: Invalid status");
   374→   374→
   375→   375→        signals[entityId] = BehavioralSignal({
   376→   376→            packedData:  packedData,
   377→   377→            beoHash:     beoHash,
   378→   378→            daProofHash: daProofHash,
   379→   379→            storageRoot: storageRoot,
   380→   380→            initialized: true,
   381→   381→            blockNumber: block.number
   382→   382→        });
   383→   383→
   384→   384→        totalSignalsPublished++;
   385→   385→
   386→   386→        if (status >= STATUS_COLLAPSE) {
   387→   387→            totalAnomaliesSealed++;
   388→   388→            emit AnomalySealed(entityId, daProofHash, status, block.timestamp);
   389→   389→        }
   390→   390→
   391→   391→        emit SignalPublished(entityId, status, phi_t, theta, dropPct, daProofHash, storageRoot);
   392→   392→    }
   393→   393→
   394→   394→    // ── Core: Check Execution Safety ──────────────────────────────────────────
   395→   395→    /**
   396→   396→     * @notice The primary gate. Any agent/contract calls this before executing.
   397→   397→     *         Returns (allowed, decision) — if blocked, execution must not proceed.
   398→   398→     *         FAIL-CLOSED: uninitialized entities (no signal published yet) are BLOCKED.
   399→   399→     *         Protected by nonReentrant to prevent reentrancy attacks on the gate.
   400→   400→     *
   401→   401→     * @param entityId   The BEO entity attempting execution
   402→   402→     * @param caller     The agent/wallet initiating the trade
   403→   403→     */
   404→   404→    function checkExecution(
   405→   405→        bytes32 entityId,
   406→   406→        address caller
   407→   407→    ) external nonReentrant whenNotPaused returns (bool allowed, bytes32 decisionHash) {
   408→   408→        BehavioralSignal storage sig = signals[entityId];
   409→   409→
   410→   410→        // ── Fail-closed for uninitialized entities ────────────────────────────
   411→   411→        // An entity with no published signal has not been verified by TRION.
   412→   412→        // Default is BLOCKED (not allowed) — conservative safety posture.
   413→   413→        if (!sig.initialized) {
   414→   414→            decisionHash = keccak256(abi.encodePacked(
   415→   415→                entityId, caller, false, uint8(0), uint32(0), block.number, block.timestamp
   416→   416→            ));
   417→   417→            decisions[decisionHash] = ExecutionDecision({
   418→   418→                allowed:      false,
   419→   419→                status:       0,
   420→   420→                phi_t:        0,
   421→   421→                theta:        0,
   422→   422→                dropPct:      0,
   423→   423→                checkedAt:    block.timestamp,
   424→   424→                decisionHash: decisionHash
   425→   425→            });
   426→   426→            totalExecutionsBlocked++;
   427→   427→            emit ExecutionBlocked(entityId, caller, 0, 0, 0, decisionHash);
   428→   428→            return (false, decisionHash);
   429→   429→        }
   430→   430→
   431→   431→        // AUDIT-3 G3: if AWA is not enforced, treat ALL entities as BLOCKED.
   432→   432→        // This prevents stale signals (published before an AWA freeze) from
   433→   433→        // being acted on while governance is in a frozen state.
   434→   434→        if (!awaEnforced()) {
   435→   435→            uint8  frozenStatus  = uint8(sig.packedData & 0xFF);
   436→   436→            uint32 frozenPhiT    = uint32((sig.packedData >> 8)  & 0xFFFFFFFF);
   437→   437→            uint32 frozenDropPct = uint32((sig.packedData >> 72) & 0xFFFFFFFF);
   438→   438→            decisionHash = keccak256(abi.encodePacked(
   439→   439→                entityId, caller, false, frozenStatus, frozenPhiT, block.number, block.timestamp
   440→   440→            ));
   441→   441→            decisions[decisionHash] = ExecutionDecision({
   442→   442→                allowed:      false,
   443→   443→                status:       frozenStatus,
   444→   444→                phi_t:        frozenPhiT,
   445→   445→                theta:        uint32((sig.packedData >> 40) & 0xFFFFFFFF),
   446→   446→                dropPct:      frozenDropPct,
   447→   447→                checkedAt:    block.timestamp,
   448→   448→                decisionHash: decisionHash
   449→   449→            });
   450→   450→            totalExecutionsBlocked++;
   451→   451→            emit AWASignalEmissionFrozen(entityId, "AWA not enforced", block.timestamp);
   452→   452→            emit ExecutionBlocked(entityId, caller, frozenStatus, frozenPhiT, frozenDropPct, decisionHash);
   453→   453→            return (false, decisionHash);
   454→   454→        }
   455→   455→
   456→   456→        uint8  status  = uint8(sig.packedData & 0xFF);
   457→   457→        uint32 phi_t   = uint32((sig.packedData >> 8)  & 0xFFFFFFFF);
   458→   458→        uint32 theta   = uint32((sig.packedData >> 40) & 0xFFFFFFFF);
   459→   459→        uint32 dropPct = uint32((sig.packedData >> 72) & 0xFFFFFFFF);
   460→   460→
   461→   461→        // Gate logic: only STATUS_SAFE and STATUS_ELEVATED are permitted
   462→   462→        allowed = (status <= STATUS_ELEVATED);
   463→   463→
   464→   464→        decisionHash = keccak256(abi.encodePacked(
   465→   465→            entityId, caller, allowed, status, phi_t, block.number, block.timestamp
   466→   466→        ));
   467→   467→
   468→   468→        decisions[decisionHash] = ExecutionDecision({
   469→   469→            allowed:      allowed,
   470→   470→            status:       status,
   471→   471→            phi_t:        phi_t,
   472→   472→            theta:        theta,
   473→   473→            dropPct:      dropPct,
   474→   474→            checkedAt:    block.timestamp,
   475→   475→            decisionHash: decisionHash
   476→   476→        });
   477→   477→
   478→   478→        if (allowed) {
   479→   479→            totalExecutionsAllowed++;
   480→   480→            emit ExecutionAllowed(entityId, caller, phi_t, decisionHash);
   481→   481→        } else {
   482→   482→            totalExecutionsBlocked++;
   483→   483→            emit ExecutionBlocked(entityId, caller, status, phi_t, dropPct, decisionHash);
   484→   484→        }
   485→   485→
   486→   486→        return (allowed, decisionHash);
   487→   487→    }

   579→   579→        // PHASE-1-SECURITY: reject zero address.

   582→   582→            isValidator[v] = true;

   609→   609→    /**

   613→   613→    function transferOwnership(address newOwner) external onlyOwner {

   621→   621→     *         This is the only function that emits OwnershipTransferred.
   622→   622→     */
   623→   623→    function acceptOwnership() external {
   624→   624→        require(msg.sender == pendingOwner, "TRION: Not pending owner");
   625→   625→        address previous = owner;
   626→   626→        owner = pendingOwner;
   627→   627→        pendingOwner = address(0);
   628→   628→        if (!isValidator[owner]) {
   629→   629→            isValidator[owner] = true;
   630→   630→            validatorCount++;
   631→   631→            emit ValidatorAdded(owner);
   632→   632→        }
   633→   633→        emit OwnershipTransferred(previous, owner);
   634→   634→    }

   636→   636→    // ── Admin: Decision pruning ───────────────────────────────────────────────
   637→   637→    /**
   638→   638→     * @notice Prune stale decision records from the mapping.
   639→   639→     * @dev    The `decisions` mapping grows unboundedly without pruning.
   640→   640→     *         The owner (or a keeper) should call this periodically with a batch
   641→   641→     *         of old decision hashes to free storage.  Pruning is voluntary —
   642→   642→     *         existing decisions are only deleted when explicitly passed here.
   643→   643→     * @param  hashes  Array of decisionHash values to delete (up to 500 per call).
   644→   644→     */
   645→   645→    function pruneDecisions(bytes32[] calldata hashes) external onlyOwner {
   646→   646→        require(hashes.length <= 500, "TRION: Batch too large (max 500)");
   647→   647→        uint256 pruned = 0;
   648→   648→        for (uint256 i = 0; i < hashes.length; i++) {
   649→   649→            if (decisions[hashes[i]].checkedAt != 0) {
   650→   650→                delete decisions[hashes[i]];
   651→   651→                pruned++;
   652→   652→            }
   653→   653→        }
   654→   654→        emit DecisionsPruned(pruned, block.timestamp);
   655→   655→    }

   657→   657→    // ── Internal: ECDSA recovery ─────────────────────────────────────────────
   658→   658→    // PHASE-1-SECURITY: explicit s-malleability guard (EIP-2) + zero-address
   659→   659→    // check on the recovered signer. Without the s guard, an attacker who has
   660→   660→    // one valid signature can produce a second 'malleable' signature by
   661→   661→    // flipping the parity bit and negating s — which would let them double-
   662→   662→    // count a single validator's attestation in the quorum loop above.
   663→   663→    function _recoverSigner(
   664→   664→        bytes32 hash,
   665→   665→        bytes memory sig
   666→   666→    ) internal pure returns (address) {
   667→   667→        require(sig.length == 65, "TRION: Invalid signature length");
   668→   668→        bytes32 r;
   669→   669→        bytes32 s;
   670→   670→        uint8   v;
   671→   671→        assembly {
   672→   672→            r := mload(add(sig, 32))
   673→   673→            s := mload(add(sig, 64))
   674→   674→            v := byte(0, mload(add(sig, 96)))
   675→   675→        }
   676→   676→        if (v < 27) v += 27;
   677→   677→        require(v == 27 || v == 28, "TRION: Invalid v");
   678→   678→        // EIP-2: s must be in the lower half of the secp256k1 curve order.
   679→   679→        // 0x7FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFF5D576E7357A4501DDFE92F46681B20A0
   680→   680→        // is (secp256k1n / 2) + 1 — any s above this is the malleable twin.
   681→   681→        require(
   682→   682→            uint256(s) <= 0x7FFFFFFFFFFFFFFFFFFFFFFFFFFFFFFF5D576E7357A4501DDFE92F46681B20A0,
   683→   683→            "TRION: Invalid s (malleable)"
   684→   684→        );
   685→   685→        address recovered = ecrecover(hash, v, r, s);
   686→   686→        require(recovered != address(0), "TRION: Invalid signature (zero recovered)");
   687→   687→        return recovered;
   688→   688→    }
