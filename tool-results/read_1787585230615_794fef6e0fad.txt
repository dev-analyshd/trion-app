    73→@dataclass
    74→class BTCPRoute:
    75→    """Complete BTCP route with all components."""
    76→    route_id: str = ""
    77→    intent: Optional[BTCPIntent] = None
    78→    source_vm: VMType = VMType.EVM
    79→    dest_vm: VMType = VMType.EVM
    80→    source_encoded: str = ""
    81→    dest_encoded: str = ""
    82→    source_gas: Optional[GasEstimate] = None
    83→    dest_gas: Optional[GasEstimate] = None
    84→    proofs: Dict[str, Any] = field(default_factory=dict)
    85→    privacy_level: PrivacyLevel = PrivacyLevel.BASIC
    86→    status: RouteStatus = RouteStatus.PENDING
    87→    total_fee: float = 0.0
    88→    # BTCP zero-bridge invariant: assets are NEVER moved across chains.
    89→    # Value stays in escrow on the source chain and is only released/reverted
    90→    # locally after the behavioral proof is verified.  This field is always
    91→    # False for a correctly-constructed BTCP route — the test suite asserts
    92→    # this invariant explicitly.
    93→    assets_bridged: bool = False
    94→    btcp_score: float = 0.0
    95→    route_type: str = "SINGLE_CHAIN"
    96→    created_at: float = field(default_factory=time.time)
    97→    updated_at: float = field(default_factory=time.time)
    98→    
    99→    def to_dict(self) -> Dict[str, Any]:
   100→        d = asdict(self)
   101→        d['source_vm'] = self.source_vm.name
   102→        d['dest_vm'] = self.dest_vm.name
   103→        d['privacy_level'] = self.privacy_level.name
   104→        d['status'] = self.status.name
   105→        d['route_type'] = self.route_type
   106→        d['assets_bridged'] = self.assets_bridged
   107→        d['btcp_score'] = self.btcp_score
   108→        if self.intent:
   109→            d['intent'] = self.intent.to_dict()
   110→        if self.source_gas:
   111→            d['source_gas'] = asdict(self.source_gas)
   112→        if self.dest_gas:
   113→            d['dest_gas'] = asdict(self.dest_gas)
   114→        return d

   136→# ── Privacy Router ──────────────────────────────────────────────────────────
   137→
   138→class PrivacyRouter:
   139→    """
   140→    Routes BTCP intents through appropriate ZK circuits based on privacy level.
   141→    
   142→    Generates the correct set of zero-knowledge proofs for each operation.
   143→    """
   144→    
   145→    def __init__(self):
   146→        self.zk = ZKProofSystem()
   147→    
   148→    def generate_proofs(
   149→        self,
   150→        intent: BTCPIntent,
   151→        privacy_level: PrivacyLevel,
   152→        behavioral_data: Optional[Dict[str, Any]] = None,
   153→    ) -> Dict[str, Any]:
   154→        """
   155→        Generate ZK proofs appropriate for the requested privacy level.
   156→        
   157→        Returns a dict mapping circuit names to their proofs.
   158→        """
   159→        proofs = {}
   160→        
   161→        # Level 1+: Always generate intent commitment
   162→        if privacy_level >= PrivacyLevel.BASIC:
   163→            intent_witness = IntentWitness(
   164→                entity_id=intent.source_address,
   165→                intent_type=intent.intent_type,
   166→                amount=intent.amount,
   167→                source_chain=intent.source_chain,
   168→                dest_chain=intent.dest_chain,
   169→                deadline=intent.deadline,
   170→                nonce=intent.nonce.to_bytes(32, 'big'),
   171→            )
   172→            proof = self.zk.generate_intent(intent_witness)
   173→            proofs["intent_commitment"] = proof.to_dict()
   174→        
   175→        # Level 2+: Add complementarity proof
   176→        if privacy_level >= PrivacyLevel.STANDARD:
   177→            # Generate dummy complementarity proof for routing
   178→            import secrets
   179→            sense = secrets.token_bytes(32)
   180→            antisense = bytes([b ^ 0xFF for b in sense])  # Perfect complement
   181→            comp_witness = ComplementarityWitness(
   182→                sense_strand=sense,
   183→                antisense_strand=antisense,
   184→                entity_id=intent.source_address,
   185→                block_number=18000000,
   186→            )
   187→            proof = self.zk.generate_complementarity(comp_witness)
   188→            proofs["complementarity"] = proof.to_dict()
   189→        
   190→        # Level 3+: Add travel rule compliance
   191→        if privacy_level >= PrivacyLevel.COMPLIANT:
   192→            tr_witness = TravelRuleWitness(
   193→                originator_id=intent.source_address,
   194→                beneficiary_id=intent.dest_address,
   195→                amount=intent.amount,
   196→                asset_address=intent.asset,
   197→                originator_verified=True,
   198→                beneficiary_verified=True,
   199→            )
   200→            proof = self.zk.generate_travel_rule(tr_witness)
   201→            proofs["travel_rule"] = proof.to_dict()
   202→        
   203→        # Level 4+: Add behavioral credential
   204→        if privacy_level >= PrivacyLevel.FULL and behavioral_data:
   205→            bc_witness = BehavioralCredentialWitness(
   206→                entity_id=intent.source_address,
   207→                coherence_score=behavioral_data.get("coherence", 0.75),
   208→                manipulation_fingerprint=behavioral_data.get("manipulation", 0.15),
   209→                liquidity_score=behavioral_data.get("liquidity", 0.80),
   210→                akashic_depth=behavioral_data.get("depth", 500.0),
   211→                threshold_coherence=0.55,
   212→                threshold_manipulation=0.30,
   213→            )
   214→            proof = self.zk.generate_behavioral_credential(bc_witness)
   215→            proofs["behavioral_credential"] = proof.to_dict()
   216→        
   217→        # Always add IAP share proof for gas fairness
   218→        iap_witness = IAPShareWitness(
   219→            entity_id=intent.source_address,
   220→            total_gas=1_000_000,
   221→            entity_gas=151000,
   222→            total_btcp_fee=int(0.01 * 10**18),
   223→            entity_share=int(0.0015 * 10**18),
   224→            num_participants=10,
   225→        )
   226→        proof = self.zk.generate_iap_share(iap_witness)
   227→        proofs["iap_share"] = proof.to_dict()
   228→        
   229→        return proofs
   230→    
   231→    def verify_proofs(self, proofs: Dict[str, Any]) -> Tuple[bool, List[str]]:
   232→        """Verify all proofs in a set."""
   233→        all_valid = True
   234→        errors = []
   235→        
   236→        from zk import ZKProof, CircuitType
   237→        
   238→        circuit_map = {
   239→            "intent_commitment": CircuitType.INTENT_COMMITMENT,
   240→            "complementarity": CircuitType.COMPLEMENTARITY,
   241→            "travel_rule": CircuitType.TRAVEL_RULE,
   242→            "behavioral_credential": CircuitType.BEHAVIORAL_CREDENTIAL,
   243→            "iap_share": CircuitType.IAP_SHARE,
   244→        }
   245→        
   246→        for name, proof_data in proofs.items():
   247→            if name not in circuit_map:
   248→                continue
   249→            
   250→            # Reconstruct ZKProof from dict
   251→            circuit_type = circuit_map[name]
   252→            proof = ZKProof(
   253→                circuit_type=circuit_type,
   254→                proof_data=proof_data.get("proof_data", {}),
   255→                public_inputs=proof_data.get("public_inputs", {}),
   256→                commitment=proof_data.get("commitment", ""),
   257→                timestamp=proof_data.get("timestamp", 0),
   258→                version=proof_data.get("version", "1.0.0"),
   259→            )
   260→            
   261→            if not self.zk.verify(proof):
   262→                all_valid = False
   263→                errors.append(f"Invalid proof: {name}")
   264→        
   265→        return all_valid, errors

   268→# ── Cross-VM Gateway ────────────────────────────────────────────────────────
   269→
   270→class CrossVMGateway:
   271→    """
   272→    Unified gateway for all cross-VM operations.
   273→    
   274→    Provides a single interface for:
   275→      - Encoding intents for any VM
   276→      - Estimating gas across VMs
   277→      - Validating addresses for any chain
   278→      - Routing intents to appropriate adapters
   279→    """
   280→    
   281→    def __init__(self):
   282→        self.factory = VMAdapterFactory
   283→    
   284→    def encode_for_chain(self, intent: BTCPIntent, chain_id: int) -> str:
   285→        """Encode an intent for a specific chain."""
   286→        adapter = self.factory.get_by_chain_id(chain_id)
   287→        if adapter is None:
   288→            raise ValueError(f"No adapter for chain_id: {chain_id}")
   289→        return adapter.encode_intent(intent)
   290→    
   291→    def estimate_chain_gas(self, intent: BTCPIntent, chain_id: int) -> GasEstimate:
   292→        """Estimate gas for executing on a specific chain."""
   293→        adapter = self.factory.get_by_chain_id(chain_id)
   294→        if adapter is None:
   295→            raise ValueError(f"No adapter for chain_id: {chain_id}")
   296→        return adapter.estimate_gas(intent)
   297→    
   298→    def validate_chain_address(self, address: str, chain_id: int) -> bool:
   299→        """Validate an address format for a specific chain."""
   300→        adapter = self.factory.get_by_chain_id(chain_id)
   301→        if adapter is None:
   302→            return False
   303→        return adapter.validate_address(address)
   304→    
   305→    def format_chain_address(self, address: str, chain_id: int) -> str:
   306→        """Format an address for a specific chain."""
   307→        adapter = self.factory.get_by_chain_id(chain_id)
   308→        if adapter is None:
   309→            return address
   310→        return adapter.format_address(address)
   311→    
   312→    def get_vm_type(self, chain_id: int) -> VMType:
   313→        """Get the VM type for a given chain."""
   314→        return CHAIN_VM_MAP.get(chain_id, VMType.EVM)
   315→    
   316→    def get_supported_chains(self) -> List[Dict[str, Any]]:
   317→        """Get all supported chains and their VM types."""
   318→        chains = []
   319→        for chain_id, vm_type in sorted(CHAIN_VM_MAP.items()):
   320→            adapter = self.factory.get_by_vm_type(vm_type)
   321→            chains.append({
   322→                "chain_id": chain_id,
   323→                "vm_type": vm_type.name,
   324→                "vm_name": adapter.name if adapter else "Unknown",
   325→                "native_token": adapter.native_token if adapter else "UNKNOWN",
   326→            })
   327→        return chains
   328→    
   329→    def list_vms(self) -> List[Dict[str, Any]]:
   330→        """List all supported VM types."""
   331→        return self.factory.list_adapters()

   334→# ── BTCP Orchestrator ───────────────────────────────────────────────────────
   335→
   336→class BTCPOrchestrator:
   337→    """
   338→    High-level BTCP orchestration coordinator.
   339→    
   340→    Ties together:
   341→      - Intent creation and validation
   342→      - VM adapter encoding
   343→      - ZK proof generation
   344→      - Gas estimation
   345→      - Route tracking and status management
   346→    
   347→    This is the main entry point for BTCP operations in the TRION engine.
   348→    """
   349→    
   350→    def __init__(self):
   351→        self.privacy_router = PrivacyRouter()
   352→        self.gateway = CrossVMGateway()
   353→        self._routes: Dict[str, BTCPRoute] = {}
   354→    
   355→    def create_route(
   356→        self,
   357→        source_chain: int,
   358→        dest_chain: int,
   359→        source_address: str,
   360→        dest_address: str,
   361→        amount: int,
   362→        asset: str,
   363→        intent_type: str = "TRANSFER",
   364→        privacy_level: PrivacyLevel = PrivacyLevel.BASIC,
   365→        deadline_offset: int = 3600,
   366→        behavioral_data: Optional[Dict[str, Any]] = None,
   367→    ) -> OrchestrationResult:
   368→        """
   369→        Create and orchestrate a complete BTCP route.
   370→        
   371→        Steps:
   372→          1. Validate addresses for both chains
   373→          2. Create standardized BTCPIntent
   374→          3. Encode intent for source and destination VMs
   375→          4. Estimate gas for both chains
   376→          5. Generate appropriate ZK proofs
   377→          6. Track route status
   378→        """
   379→        start_time = time.perf_counter()
   380→        errors = []
   381→        
   382→        # Step 1: Validate addresses
   383→        if not self.gateway.validate_chain_address(source_address, source_chain):
   384→            # Try formatting first
   385→            source_address = self.gateway.format_chain_address(source_address, source_chain)
   386→            if not self.gateway.validate_chain_address(source_address, source_chain):
   387→                errors.append(f"Invalid source address for chain {source_chain}")
   388→        
   389→        if not self.gateway.validate_chain_address(dest_address, dest_chain):
   390→            dest_address = self.gateway.format_chain_address(dest_address, dest_chain)
   391→            if not self.gateway.validate_chain_address(dest_address, dest_chain):
   392→                errors.append(f"Invalid dest address for chain {dest_chain}")
   393→        
   394→        # Step 2: Create intent
   395→        intent_id = hashlib.sha3_256(
   396→            f"{source_chain}:{dest_chain}:{source_address}:{dest_address}:{amount}:{time.time()}".encode()
   397→        ).hexdigest()[:16]
   398→        
   399→        intent = BTCPIntent(
   400→            intent_id=f"btcp_{intent_id}",
   401→            source_chain=source_chain,
   402→            dest_chain=dest_chain,
   403→            source_address=source_address,
   404→            dest_address=dest_address,
   405→            amount=amount,
   406→            asset=asset,
   407→            intent_type=intent_type,
   408→            deadline=int(time.time()) + deadline_offset,
   409→            nonce=int(time.time() * 1000) % (2**32),
   410→        )
   411→        
   412→        # Step 3: Encode for both VMs
   413→        source_vm = self.gateway.get_vm_type(source_chain)
   414→        dest_vm = self.gateway.get_vm_type(dest_chain)
   415→        
   416→        try:
   417→            source_encoded = self.gateway.encode_for_chain(intent, source_chain)
   418→        except Exception as e:
   419→            source_encoded = ""
   420→            errors.append(f"Source encoding failed: {str(e)[:80]}")
   421→        
   422→        try:
   423→            dest_encoded = self.gateway.encode_for_chain(intent, dest_chain)
   424→        except Exception as e:
   425→            dest_encoded = ""
   426→            errors.append(f"Dest encoding failed: {str(e)[:80]}")
   427→        
   428→        # Step 4: Estimate gas
   429→        try:
   430→            source_gas = self.gateway.estimate_chain_gas(intent, source_chain)
   431→        except Exception as e:
   432→            source_gas = None
   433→            errors.append(f"Source gas estimate failed: {str(e)[:80]}")
   434→        
   435→        try:
   436→            dest_gas = self.gateway.estimate_chain_gas(intent, dest_chain)
   437→        except Exception as e:
   438→            dest_gas = None
   439→            errors.append(f"Dest gas estimate failed: {str(e)[:80]}")
   440→        
   441→        total_fee = 0.0
   442→        if source_gas:
   443→            total_fee += source_gas.estimated_fee
   444→        if dest_gas:
   445→            total_fee += dest_gas.estimated_fee
   446→        
   447→        # Step 5: Generate ZK proofs
   448→        try:
   449→            proofs = self.privacy_router.generate_proofs(intent, privacy_level, behavioral_data)
   450→            proof_names = list(proofs.keys())
   451→        except Exception as e:
   452→            proofs = {}
   453→            proof_names = []
   454→            errors.append(f"Proof generation failed: {str(e)[:80]}")
   455→        
   456→        # Step 6: Create and track route
   457→        route = BTCPRoute(
   458→            route_id=f"route_{intent_id}",
   459→            intent=intent,
   460→            source_vm=source_vm,
   461→            dest_vm=dest_vm,
   462→            source_encoded=source_encoded,
   463→            dest_encoded=dest_encoded,
   464→            source_gas=source_gas,
   465→            dest_gas=dest_gas,
   466→            proofs=proofs,
   467→            privacy_level=privacy_level,
   468→            status=RouteStatus.PROOFS_GENERATED if proofs else RouteStatus.INTENT_CREATED,
   469→            total_fee=total_fee,
   470→        )
   471→        
   472→        self._routes[route.route_id] = route
   473→        
   474→        execution_time = (time.perf_counter() - start_time) * 1000
   475→        
   476→        success = len(errors) == 0
   477→        
   478→        return OrchestrationResult(
   479→            success=success,
   480→            route=route,
   481→            proofs_generated=proof_names,
   482→            errors=errors,
   483→            execution_time_ms=execution_time,
   484→        )
   485→    
   486→    def get_route(self, route_id: str) -> Optional[BTCPRoute]:
   487→        """Get a route by ID."""
   488→        return self._routes.get(route_id)
   489→    
   490→    def update_route_status(self, route_id: str, status: RouteStatus) -> bool:
   491→        """Update the status of a route."""
   492→        route = self._routes.get(route_id)
   493→        if route is None:
   494→            return False
   495→        route.status = status
   496→        route.updated_at = time.time()
   497→        return True
   498→    
   499→    def list_routes(self) -> List[Dict[str, Any]]:
   500→        """List all tracked routes."""
   501→        return [
   502→            {
   503→                "route_id": r.route_id,
   504→                "source_chain": r.intent.source_chain if r.intent else 0,
   505→                "dest_chain": r.intent.dest_chain if r.intent else 0,
   506→                "source_vm": r.source_vm.name,
   507→                "dest_vm": r.dest_vm.name,
   508→                "status": r.status.name,
   509→                "privacy_level": r.privacy_level.name,
   510→                "total_fee": r.total_fee,
   511→                "created_at": r.created_at,
   512→            }
   513→            for r in self._routes.values()
   514→        ]
   515→    
   516→    def verify_route_proofs(self, route_id: str) -> Tuple[bool, List[str]]:
   517→        """Verify all proofs for a route."""
   518→        route = self._routes.get(route_id)
   519→        if route is None:
   520→            return False, ["Route not found"]
   521→        
   522→        return self.privacy_router.verify_proofs(route.proofs)

   525→# ── Proof Aggregator ────────────────────────────────────────────────────────
   526→
   527→class ProofAggregator:
   528→    """
   529→    Collects and aggregates proofs across multiple chains.
   530→    
   531→    Provides:
   532→      - Multi-chain proof aggregation
   533→      - Merkle root of all proofs
   534→      - Aggregate verification
   535→      - Proof availability tracking
   536→    """
   537→    
   538→    def __init__(self):
   539→        self._proofs: Dict[str, Dict[str, Any]] = {}
   540→        self._merkle_leaves: List[bytes] = []
   541→    
   542→    def add_proof(self, proof_id: str, proof_data: Dict[str, Any], chain_id: int) -> str:
   543→        """Add a proof to the aggregator."""
   544→        leaf = hashlib.sha3_256(
   545→            f"{proof_id}:{chain_id}:{json.dumps(proof_data, sort_keys=True)}".encode()
   546→        ).digest()
   547→        
   548→        self._proofs[proof_id] = {
   549→            "proof": proof_data,
   550→            "chain_id": chain_id,
   551→            "leaf": leaf.hex(),
   552→            "index": len(self._merkle_leaves),
   553→            "timestamp": time.time(),
   554→        }
   555→        self._merkle_leaves.append(leaf)
   556→        
   557→        return leaf.hex()
   558→    
   559→    def aggregate(self) -> Dict[str, Any]:
   560→        """Compute the aggregate Merkle root of all proofs."""
   561→        from zk import merkle_root
   562→        
   563→        if not self._merkle_leaves:
   564→            return {
   565→                "proof_count": 0,
   566→                "merkle_root": hashlib.sha3_256(b"empty").hexdigest(),
   567→            }
   568→        
   569→        root = merkle_root(self._merkle_leaves)
   570→        
   571→        return {
   572→            "proof_count": len(self._merkle_leaves),
   573→            "merkle_root": root.hex(),
   574→            "chains": list(set(p["chain_id"] for p in self._proofs.values())),
   575→        }
   576→
