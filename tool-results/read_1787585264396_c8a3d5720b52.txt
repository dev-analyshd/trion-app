   168→export function DataTable({
   169→  headers,
   170→  rows,
   171→  maxHeight = 320,
   172→  emptyMessage = 'Loading...',
   173→  sortable = false,
   174→  exportable = false,
   175→  exportName = 'trion_export',
   176→  onRowClick,
   177→  copyableColumns,
   178→}: {
   179→  headers: string[];
   180→  rows: (string | React.ReactNode)[][];
   181→  maxHeight?: number;
   182→  emptyMessage?: string;
   183→  sortable?: boolean;
   184→  exportable?: boolean;
   185→  exportName?: string;
   186→  onRowClick?: (rowIdx: number) => void;
   187→  copyableColumns?: number[]; // column indices that should show a copy button
   188→}) {
   189→  const [sortCol, setSortCol] = useState<number | null>(null);
   190→  const [sortDir, setSortDir] = useState<'asc' | 'desc' | null>(null);
   191→
   192→  // Apply sort
   193→  let displayRows = rows;
   194→  if (sortable && sortCol !== null && sortDir !== null) {
   195→    displayRows = [...rows].sort((a, b) => {
   196→      const av = a[sortCol];
   197→      const bv = b[sortCol];
   198→      // Try numeric compare first
   199→      const an = typeof av === 'string' ? parseFloat(av.replace(/[^0-9.-]/g, '')) : NaN;
   200→      const bn = typeof bv === 'string' ? parseFloat(bv.replace(/[^0-9.-]/g, '')) : NaN;
   201→      if (!isNaN(an) && !isNaN(bn)) {
   202→        return sortDir === 'asc' ? an - bn : bn - an;
   203→      }
   204→      const as = String(av ?? '');
   205→      const bs = String(bv ?? '');
   206→      return sortDir === 'asc' ? as.localeCompare(bs) : bs.localeCompare(as);
   207→    });
   208→  }
   209→
   210→  const handleSort = (colIdx: number) => {
   211→    if (!sortable) return;
   212→    if (sortCol === colIdx) {
   213→      // Cycle: asc {'->'} desc {'->'} null
   214→      if (sortDir === 'asc') setSortDir('desc');
   215→      else if (sortDir === 'desc') { setSortCol(null); setSortDir(null); }
   216→    } else {
   217→      setSortCol(colIdx);
   218→      setSortDir('asc');
   219→    }
   220→  };
   221→
   222→  const exportCSV = () => {
   223→    const csv = [
   224→      headers.join(','),
   225→      ...rows.map(r => r.map(c => {
   226→        const s = typeof c === 'string' ? c : String(c ?? '');
   227→        return `"${s.replace(/"/g, '""')}"`;
   228→      }).join(',')),
   229→    ].join('\n');
   230→    const blob = new Blob([csv], { type: 'text/csv' });
   231→    const url = URL.createObjectURL(blob);
   232→    const a = document.createElement('a');
   233→    a.href = url;
   234→    a.download = `${exportName}.csv`;
   235→    a.click();
   236→    URL.revokeObjectURL(url);
   237→  };
   238→
   239→  const exportJSON = () => {
   240→    const data = rows.map(r => {
   241→      const obj: Record<string, any> = {};
   242→      headers.forEach((h, i) => { obj[h] = r[i]; });
   243→      return obj;
   244→    });
   245→    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
   246→    const url = URL.createObjectURL(blob);
   247→    const a = document.createElement('a');
   248→    a.href = url;
   249→    a.download = `${exportName}.json`;
   250→    a.click();
   251→    URL.revokeObjectURL(url);
   252→  };
   253→
   254→  const copyCell = (val: React.ReactNode) => {
   255→    const s = typeof val === 'string' ? val : String(val ?? '');
   256→    navigator.clipboard?.writeText(s).catch(() => {});
   257→  };
   258→
   259→  return (
   260→    <div>
   261→      {exportable && rows.length > 0 && (
   262→        <div className="flex justify-end gap-2 mb-2">
   263→          <button
   264→            onClick={exportCSV}
   265→            className="text-xs px-2 py-1 rounded border border-border hover:bg-accent text-muted-foreground hover:text-foreground transition-colors"
   266→            aria-label="Export as CSV"
   267→          >
   268→            down CSV
   269→          </button>
   270→          <button
   271→            onClick={exportJSON}
   272→            className="text-xs px-2 py-1 rounded border border-border hover:bg-accent text-muted-foreground hover:text-foreground transition-colors"
   273→            aria-label="Export as JSON"
   274→          >
   275→            down JSON
   276→          </button>
   277→        </div>
   278→      )}
   279→      <div className="overflow-auto" style={{ maxHeight }}>
   280→        <table className="w-full text-sm">
   281→          <thead className="bg-muted/50 sticky top-0 backdrop-blur">
   282→            <tr>
   283→              {headers.map((h, i) => (
   284→                <th
   285→                  key={i}
   286→                  onClick={() => handleSort(i)}
   287→                  className={`text-left p-2 md:p-3 text-xs font-semibold text-muted-foreground whitespace-nowrap ${
   288→                    sortable ? 'cursor-pointer hover:text-foreground select-none' : ''
   289→                  }`}
   290→                  aria-sort={sortCol === i ? (sortDir === 'asc' ? 'ascending' : sortDir === 'desc' ? 'descending' : 'none') : undefined}
   291→                >
   292→                  {h}
   293→                  {sortable && sortCol === i && (
   294→                    <span className="ml-1" aria-hidden>
   295→                      {sortDir === 'asc' ? 'up' : sortDir === 'desc' ? 'down' : ''}
   296→                    </span>
   297→                  )}
   298→                </th>
   299→              ))}
   300→            </tr>
   301→          </thead>
   302→          <tbody>
   303→            {displayRows.length === 0 ? (
   304→              <tr>
   305→                <td colSpan={headers.length} className="p-8 text-center text-muted-foreground">
   306→                  {emptyMessage}
   307→                </td>
   308→              </tr>
   309→            ) : (
   310→              displayRows.map((row, i) => (
   311→                <tr
   312→                  key={i}
   313→                  className={`border-b border-border/50 hover:bg-muted/30 transition-colors ${
   314→                    onRowClick ? 'cursor-pointer' : ''
   315→                  }`}
   316→                  onClick={onRowClick ? () => onRowClick(i) : undefined}
   317→                >
   318→                  {row.map((cell, j) => (
   319→                    <td key={j} className="p-2 md:p-3 whitespace-nowrap max-w-[240px]">
   320→                      <div className="flex items-center gap-1.5 overflow-hidden">
   321→                        <span className="truncate">{cell}</span>
   322→                        {copyableColumns?.includes(j) && cell && (
   323→                          <button
   324→                            onClick={(e) => { e.stopPropagation(); copyCell(cell); }}
   325→                            className="text-[10px] opacity-30 hover:opacity-100 transition-opacity text-muted-foreground"
   326→                            title="Copy to clipboard"
   327→                            aria-label="Copy value"
   328→                          >
   329→                            ⧉
   330→                          </button>
   331→                        )}
   332→                      </div>
   333→                    </td>
   334→                  ))}
   335→                </tr>
   336→              ))
   337→            )}
   338→          </tbody>
   339→        </table>
   340→      </div>
   341→    </div>
   342→  );
   343→}
