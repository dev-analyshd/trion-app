    75→                ]
    76→                # Cap evictions per sweep to avoid holding the lock too long
    77→                for k in stale_keys[:2000]:
    78→                    del _rl_buckets[k]
    79→                    evicted += 1

    83→            _log.warning("rate-limiter cleanup error: %s", e)

    87→_rl_cleanup_thread = threading.Thread(
    88→    target=_rl_cleanup_loop, name="rl-cleanup", daemon=True

    93→@app.before_request
    94→def _rate_limit():
    95→    # Allow health probes and static files through without counting
    96→    path = request.path
    97→    if path in ("/api/v1/health", "/favicon.ico") or path.startswith("/static/"):
    98→        return None
    99→
   100→    ip  = _get_client_ip()
   101→    now = time.time()
   102→
   103→    with _rl_lock:
   104→        bucket = _rl_buckets.setdefault(ip, deque())
   105→        # Evict timestamps outside the current window
   106→        cutoff = now - _RL_WINDOW
   107→        while bucket and bucket[0] < cutoff:
   108→            bucket.popleft()
   109→
   110→        if len(bucket) >= _RL_MAX_REQS:
   111→            retry_after = int(_RL_WINDOW - (now - bucket[0])) + 1
   112→            _log.warning("Rate limit exceeded: ip=%s requests=%d window=%ds", ip, len(bucket), _RL_WINDOW)
   113→            from flask import Response
   114→            return Response(
   115→                json.dumps({"error": "rate_limit_exceeded",
   116→                            "message": f"Too many requests — limit is {_RL_MAX_REQS} per {_RL_WINDOW}s",
   117→                            "retry_after_seconds": retry_after}),
   118→                status=429,
   119→                mimetype="application/json",
   120→                headers={"Retry-After": str(retry_after)},
   121→            )
   122→
   123→        bucket.append(now)
   124→
   125→        # Hard cap: if too many unique IPs tracked, evict oldest ones synchronously
   126→        # (fallback to the background thread for normal operation)
   127→        if len(_rl_buckets) > _RL_BUCKET_CAP:
   128→            stale_cutoff = now - _RL_STALE_AFTER
   129→            stale_keys = [k for k, v in _rl_buckets.items() if not v or v[-1] < stale_cutoff]
   130→            for k in stale_keys[:1000]:
   131→                del _rl_buckets[k]
   132→
   133→    return None

   463→# ── Behavioral plane computation — hash-seeded + live FAISS enrichment ───────
   464→# Hash gives stable deterministic base values per entity.
   465→# FAISS enrichment overwrites Mental and ANIMA with live indexed data
   466→# when real vectors exist for the entity (non-neutral-prior).
   467→
   468→def _entity_seed(eid: str) -> float:
   469→    # SHA3-256 — matches canonical BH L0.1 spec, Rust trion_common::bh_id(),
   470→    # and TypeScript entityIdFromAddr() in chains/shared/canonical_bh.ts.
   471→    # Using SHA-256 here was a bug: it broke cross-language BEO identity consistency.
   472→    h = hashlib.sha3_256(eid.encode()).digest()
   473→    return int.from_bytes(h[:4], "big") / 0xFFFFFFFF
   474→
   475→# Minimum coherence threshold used in COLD_START SILENCE signals ────────────
   476→# Mirrors THETA_MIN from src/core/coherence_engine.py (0.55) so COLD_START
   477→# responses carry a meaningful threshold field without importing the engine.
   478→THETA_MIN_COLD_START: float = 0.55
   479→
   480→# FAISS enrichment cache  ──────────────────────────────────────────────────
   481→# Phase 2.2: Bounded LRU cache with TTL + thread safety.
   482→# Replaces the unbounded dict that could grow without limit under sustained
   483→# traffic. Uses functools.lru_cache with a time-bucketed key to force refresh
   484→# every _FAISS_PLANE_TTL seconds.
   485→import threading as _threading
   486→from functools import lru_cache as _lru_cache
   487→
   488→_faiss_lock = _threading.Lock()
   489→_FAISS_PLANE_TTL = 45  # seconds — also drives the time-bucket key
   490→_FAISS_CACHE_MAX = 10_000  # bound on cached entries (per TTL bucket)
   491→
   492→@_lru_cache(maxsize=_FAISS_CACHE_MAX)
   493→def _query_faiss_planes_cached(eid: str, ts_bucket: int) -> tuple:
   494→    """
   495→    Cached wrapper. ts_bucket changes every _FAISS_PLANE_TTL seconds, forcing
   496→    a fresh fetch. Returns a JSON-serialisable tuple (None or dict-as-tuple-of-pairs)
   497→    so lru_cache can hash it.
   498→    """
   499→    import urllib.request as _ur
   500→    try:
   501→        # ── Mental confidence ──────────────────────────────────────────────
   502→        with _ur.urlopen(
   503→            f"http://127.0.0.1:8000/api/v1/mental_confidence/{eid}", timeout=1
   504→        ) as _r:
   505→            mental = json.loads(_r.read())
   506→        if mental.get("status") == "neutral_prior":
   507→            return None
   508→        # Only skip when BOTH history is empty AND mental score is exactly the
   509→        # neutral prior (0.5). A valid archetype_id of -1 just means unclassified
   510→        # — ANIMA can still provide a meaningful score in that state.
   511→        _m_val_raw = float(mental.get("mental_m", 0.5))
   512→        if mental.get("history_window", 0) == 0 and abs(_m_val_raw - 0.5) < 1e-6:
   513→            return None
   514→        m_val = float(mental.get("mental_m", 0.5))
   515→
   516→        # ── ANIMA score ────────────────────────────────────────────────────
   517→        with _ur.urlopen(
   518→            f"http://127.0.0.1:8000/api/v1/anima/{eid}", timeout=1
   519→        ) as _r:
   520→            anima_d = json.loads(_r.read())
   521→        a_val = float(anima_d.get("anima_score", 0.5))
   522→
   523→        # ── Depth (physical proxy) ─────────────────────────────────────────
   524→        with _ur.urlopen(
   525→            f"http://127.0.0.1:8000/api/v1/depth/{eid}", timeout=1
   526→        ) as _r:
   527→            depth_d = json.loads(_r.read())
   528→        depth = float(depth_d.get("akashic_depth", 0.0))
   529→        phi_live = min(1.0, 0.40 + 0.55 * depth) if depth > 0 else None
   530→
   531→        return ({"m": m_val, "anima": a_val, "phi_live": phi_live,
   532→                 "akashic_depth": depth}, time.time())
   533→    except Exception:
   534→        return (None, time.time())

   537→def _query_faiss_planes(eid: str) -> dict | None:
   538→    """
   539→    Query the FAISS engine (port 8000) for live plane values.
   540→    Returns a dict with keys 'm', 'anima', 'phi_live' if real (non-neutral)
   541→    data exists, otherwise None so the caller falls back to hash values.
   542→
   543→    Thread-safe via _faiss_lock; bounded via lru_cache(maxsize=10000).
   544→    TTL enforced via time-bucketed cache key (refreshes every 45s).
   545→    """
   546→    ts_bucket = int(time.time() // _FAISS_PLANE_TTL)
   547→    with _faiss_lock:
   548→        cached = _query_faiss_planes_cached(eid, ts_bucket)
   549→    if cached is None:
   550→        return None
   551→    result, _ts = cached
   552→    return result
