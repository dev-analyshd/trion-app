   100→    Python's built-in hash() is salted per-process and would change on restart."""
   101→    import hashlib

   388→def compute_bridge_pairs_eliminated(n: int) -> int:
   389→    """Bridge pairs eliminated = N × (N-1) / 2."""
   390→    return n * (n - 1) // 2

   393→def get_bootstrap_status(chains: List[ChainConfig]) -> dict:
   394→    """Get the current bootstrap status across all phases."""
   395→    phase_counts: Dict[BootstrapPhase, int] = {}
   396→    vm_counts: Dict[VmFamily, int] = {}
   397→    live_count = 0
   398→
   399→    for c in chains:
   400→        phase_counts[c.btcp_phase] = phase_counts.get(c.btcp_phase, 0) + 1
   401→        vm_counts[c.vm_family] = vm_counts.get(c.vm_family, 0) + 1
   402→        if c.is_live:
   403→            live_count += 1
   404→
   405→    # Compute cumulative bridge pairs eliminated at each phase
   406→    phase_pairs = {}
   407→    cumulative = 0
   408→    for phase in sorted(phase_counts.keys()):
   409→        cumulative += phase_counts[phase]
   410→        phase_pairs[phase.name] = {
   411→            "chains_added": phase_counts[phase],
   412→            "cumulative_chains": cumulative,
   413→            "pairs_eliminated": compute_bridge_pairs_eliminated(cumulative),
   414→        }
   415→
   416→    return {
   417→        "total_chains": len(chains),
   418→        "live_chains": live_count,
   419→        "vm_families": len(vm_counts),
   420→        "phase_distribution": {p.name: c for p, c in phase_counts.items()},
   421→        "vm_distribution": {v.name: c for v, c in vm_counts.items()},
   422→        "bridge_pairs_eliminated": compute_bridge_pairs_eliminated(len(chains)),
   423→        "phase_progression": phase_pairs,
   424→        "validator_coverage_tiers": {
   425→            "tier_1_1_5_chains": "1.0× BASE_STAKE",
   426→            "tier_2_6_20_chains": "2.5× BASE_STAKE",
   427→            "tier_3_21_50_chains": "5.0× BASE_STAKE",
   428→            "tier_4_51_plus_chains": "10.0× BASE_STAKE",
   429→        },
   430→    }
