     1→#!/usr/bin/env node
     2→/**
     3→ * TRION Multi-Chain Relayer
     4→ * =========================
     5→ * Polls the Akashic Oracle for the latest signal on each monitored entity,
     6→ * packs it into the bit layout expected by TRIONOracleV3.publishSignal, and
     7→ * either signs+sends to every configured chain or runs in DRY_RUN mode.
     8→ *
     9→ * Bit layout (publishSignal packedData, uint256):
    10→ *   bits   0..7    : status   (uint8 — 1 = SAFE, 0 = COLLAPSE_INTERCEPTED)
    11→ *   bits   8..39   : coherence (uint32 — coherence × 1e6, capped at 2^32-1)
    12→ *   bits  40..71   : threshold (uint32 — threshold × 1e6, capped at 2^32-1)
    13→ *   bits  72..135  : block_num (uint64)
    14→ *   bits 136..199  : timestamp (uint64 — unix seconds)
    15→ *
    16→ * Quorum: publishSignal requires `quorumRequired` signatures over
    17→ *   keccak256(abi.encodePacked(chainId, oracleAddr, txId, packedData))
    18→ * each prefixed with the EIP-191 "\x19Ethereum Signed Message:\n32" header.
    19→ * For a single-validator setup this relayer signs with RELAYER_PRIVATE_KEY
    20→ * (which must also be the registered validator). For multi-validator quorum,
    21→ * collect signatures from peers via SIGNER_KEYS_JSON (comma separated).
    22→ *
    23→ * Required env (live mode):
    24→ *   RELAYER_PRIVATE_KEY      hex private key (0x… or 64 hex chars) for a registered validator
    25→ *   ORACLE_API_URL           http://127.0.0.1:5000          (default)
    26→ *   MONITORED_ENTITIES       comma-separated list (default: TRION oracle contract)
    27→ *   POLL_INTERVAL_MS         default 30000
    28→ *   ARBITRUM_ORACLE_ADDR     contract addr on Arbitrum One
    29→ *   ETH_MAINNET_ORACLE_ADDR  contract addr on Ethereum
    30→ *   POLYGON_ORACLE_ADDR      contract addr on Polygon
    31→ *   OPTIMISM_ORACLE_ADDR     ...
    32→ *   BASE_ORACLE_ADDR         ...
    33→ *   BNB_ORACLE_ADDR          ...
    34→ *   AVALANCHE_ORACLE_ADDR    ...
    35→ *   LINEA_ORACLE_ADDR        ...
    36→ *   SCROLL_ORACLE_ADDR       ...
    37→ *   ZKSYNC_ORACLE_ADDR       ...
    38→ *
    39→ * If RELAYER_PRIVATE_KEY is missing the relayer runs in DRY_RUN mode and
    40→ * logs the would-be transaction for every chain. If a per-chain *_ORACLE_ADDR
    41→ * is missing, that chain is skipped.
    42→ */

   229→function classifyGateStatus(coherence, threshold) {
   230→  const ratio = coherence / (threshold || 0.55);
   231→  if (ratio >= 1.05) return 1n; // SAFE
   232→  if (ratio >= 0.90) return 2n; // ELEVATED
   233→  if (ratio >= 0.70) return 3n; // COLLAPSE
   234→  return 4n;                    // HOSTILE
   235→}

   237→function packGateSignal(signal, blockNum) {
   238→  const phi_t   = signal.coherence || signal.signal_value || 0.5;
   239→  const theta   = signal.threshold || 0.55;
   240→  const status  = classifyGateStatus(phi_t, theta);
   241→  const drop    = BigInt(Math.max(0, Math.floor((theta - phi_t) / theta * 100 * 10000)));
   242→  const phi_t32 = clampU32(phi_t * 1_000_000);
   243→  const theta32 = clampU32(theta * 1_000_000);
   244→  const block64 = clampU64(blockNum);
   245→  const ts64    = clampU64(Math.floor(Date.now() / 1000));
   246→
   247→  return {
   248→    packed: (ts64 << 168n) | (block64 << 104n) | (drop << 72n)
   249→          | (theta32 << 40n) | (phi_t32 << 8n) | status,
   250→    status: Number(status),
   251→    phi_t, theta,
   252→    statusLabel: ["","SAFE","ELEVATED","COLLAPSE","HOSTILE"][Number(status)] || "UNKNOWN",
   253→  };
   254→}

   347→// ── Signal packing ───────────────────────────────────────────────────────────
   348→function clampU32(x) {
   349→  if (x < 0) return 0n;
   350→  const v = BigInt(Math.floor(x));
   351→  const max = (1n << 32n) - 1n;
   352→  return v > max ? max : v;
   353→}
   354→
   355→function clampU64(x) {
   356→  if (x < 0) return 0n;
   357→  const v = BigInt(Math.floor(x));
   358→  const max = (1n << 64n) - 1n;
   359→  return v > max ? max : v;
   360→}

   362→/**
   363→ * Pack a TRION signal into the uint256 layout expected by TRIONOracleV3.
   364→ *  status[8] | coherence[32] | threshold[32] | blockNum[64] | timestamp[64]
   365→ *
   366→ * Oracle v2 response uses `coherence_score` (not `coherence` / `signal_value`).
   367→ * We support both field names for forward/backward compatibility.
   368→ */
   369→function packSignal(signal) {
   370→  const coh = signal.coherence_score ?? signal.coherence ?? signal.signal_value ?? 0.5;
   371→  const thr = signal.threshold ?? 0.55;
   372→  const isSafe = coh >= thr ? 1n : 0n;
   373→  const coherence = clampU32(coh * 1_000_000);
   374→  const threshold = clampU32(thr * 1_000_000);
   375→  const blockNum  = clampU64(signal.block_num || Date.now() / 1000 | 0);
   376→  const timestamp = clampU64(Math.floor(Date.now() / 1000));
   377→
   378→  return (timestamp << 136n)
   379→       | (blockNum  << 72n)
   380→       | (threshold << 40n)
   381→       | (coherence << 8n)
   382→       | isSafe;
   383→}

   385→function deriveTxId(entity, signal) {
   386→  // bytes32 = exactly 64 hex chars. Oracle's signal_id may be a UUID (with
   387→  // dashes) or a hex string. Strip dashes so UUIDs become pure hex.
   388→  const rawSig  = (signal.signal_id || "").replace(/^0x/, "").replace(/-/g, "").toLowerCase();
   389→  // If after stripping dashes the result is pure hex, use it; otherwise hash it.
   390→  const sigHex  = /^[0-9a-f]+$/.test(rawSig)
   391→    ? rawSig
   392→    : ethers.keccak256(ethers.toUtf8Bytes(signal.signal_id || "")).slice(2);
   393→  const sigPart = sigHex.padEnd(32, "0").slice(0, 32); // 16 bytes = 32 hex
   394→  const ts      = Math.floor(Date.now() / 1000).toString(16).padStart(16, "0");
   395→  // Entity may be a hex address (0x…) OR a string like "uniswap"/"aave".
   396→  const raw = (entity || "").replace(/^0x/, "").toLowerCase();
   397→  const isHex = /^[0-9a-f]+$/.test(raw);
   398→  const entHex  = isHex
   399→    ? raw.padStart(40, "0")
   400→    : ethers.keccak256(ethers.toUtf8Bytes(entity || "")).slice(2); // 64 hex
   401→  const entTail = entHex.slice(-16); // last 8 bytes
   402→  const hex     = (sigPart + ts + entTail).padEnd(64, "0").slice(0, 64);
   403→  return "0x" + hex;
   404→}

   423→async function pushToChain(chain, entity, signal, wallet) {
   424→  const txId = deriveTxId(entity, signal);
   425→  const packed = packSignal(signal);
   426→
   427→  if (DRY_RUN || !chain.addr) {
   428→    const reason = DRY_RUN ? "DRY_RUN (no RELAYER_PRIVATE_KEY)" : `no ${chain.addrEnv}`;
   429→    console.log(`  [${chain.key.padEnd(10)}] ${reason} — would publishSignal(txId=${txId.slice(0,18)}…, packed=0x${packed.toString(16).slice(0,16)}…)`);
   430→    relayerState.chains[chain.key] = {
   431→      chain_id: chain.chainId, oracle_address: chain.addr || null,
   432→      mode: "DRY_RUN", reason, last_tx_id: txId, updated_at: new Date().toISOString(),
   433→    };
   434→    persistRelayerState();
   435→    return { ok: true, dry: true };
   436→  }
   437→
   438→  try {
   439→    const provider = new ethers.JsonRpcProvider(chain.rpc, chain.chainId);
   440→    const signer = wallet.connect(provider);
   441→    const oracle = new ethers.Contract(chain.addr, ABI, signer);
   442→
   443→    // Build the digest the contract reconstructs in publishSignal:
   444→    //   ethSignedMessageHash(keccak256(abi.encodePacked(chainId, oracleAddr, txId, packedData)))
   445→    const inner = ethers.solidityPackedKeccak256(
   446→      ["uint256", "address", "bytes32", "uint256"],
   447→      [chain.chainId, chain.addr, txId, packed]
   448→    );
   449→    // ethers v6 signMessage handles the EIP-191 prefix when given raw bytes.
   450→    const sig = await signer.signMessage(ethers.getBytes(inner));
   451→
   452→    // Submit to chain with retry on transient errors (network / RPC drops).
   453→    // 4xx contract reverts and quorum failures are NOT retried.
   454→    const receipt = await withRetry(`push:${chain.key}`, async () => {
   455→      const tx = await oracle.publishSignal(txId, packed, [sig]);
   456→      return await tx.wait(1);
   457→    });
   458→    console.log(`  [${chain.key.padEnd(10)}] published txId=${txId.slice(0,18)}… block=${receipt.blockNumber} hash=${receipt.hash}`);
   459→    const prev = relayerState.chains[chain.key] || {};
   460→    relayerState.chains[chain.key] = {
   461→      ...prev,
   462→      chain_id: chain.chainId, oracle_address: chain.addr,
   463→      mode: "REAL", last_status: "ok",
   464→      last_tx_id: txId,
   465→      last_tx_hash: receipt.hash,
   466→      last_block: receipt.blockNumber,
   467→      last_real_tx_hash: receipt.hash,           // preserved across later failures
   468→      last_real_block: receipt.blockNumber,
   469→      last_real_at: new Date().toISOString(),
   470→      last_signal_value: signal.signal_value, last_signal_type: signal.signal_type,
   471→      updated_at: new Date().toISOString(),
   472→      last_error: null,
   473→    };
   474→    persistRelayerState();
   475→    return { ok: true, hash: receipt.hash };
   476→  } catch (e) {
   477→    const msg = e?.shortMessage || e?.message || String(e);
   478→    console.error(`  [${chain.key.padEnd(10)}] FAILED: ${msg}`);
   479→    const prev = relayerState.chains[chain.key] || {};
   480→    // If we had a prior successful publish, mark DEGRADED (latest attempt failed);
   481→    // otherwise REJECTED. Either way, last_real_* fields are preserved.
   482→    const mode = prev.last_real_tx_hash ? "DEGRADED" : "REJECTED";
   483→    relayerState.chains[chain.key] = {
   484→      ...prev,
   485→      chain_id: chain.chainId, oracle_address: chain.addr,
   486→      mode, last_status: "error", last_error: msg,
   487→      last_attempt_at: new Date().toISOString(),
   488→    };
   489→    persistRelayerState();
   490→    return { ok: false, error: msg };
   491→  }
   492→}

   494→// ── Retry helper ─────────────────────────────────────────────────────────────
   495→// Exponential-backoff retry for transient network failures.  Distinguishes
   496→// retryable errors (timeouts, 5xx, network drops) from permanent ones (4xx
   497→// auth / contract revert) so we don't waste cycles hammering a broken endpoint.
   498→const MAX_RETRIES    = parseInt(process.env.RELAYER_MAX_RETRIES    || "3",    10);
   499→const RETRY_BASE_MS = parseInt(process.env.RELAYER_RETRY_BASE_MS || "500", 10);
   500→
   501→function isRetryableError(err) {
   502→  if (!err) return false;
   503→  // Network / timeout errors are retryable
   504→  if (err.code === "ECONNABORTED" || err.code === "ETIMEDOUT" ||
   505→      err.code === "ENOTFOUND"   || err.code === "ECONNRESET") return true;
   506→  // 5xx server errors are retryable
   507→  const status = err.response?.status;
   508→  if (status && status >= 500 && status < 600) return true;
   509→  // 429 Too Many Requests is retryable
   510→  if (status === 429) return true;
   511→  // Contract reverts / 4xx are NOT retryable
   512→  return false;
   513→}
   514→
   515→async function withRetry(label, fn) {
   516→  let lastErr = null;
   517→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   518→    try {
   519→      return await fn(attempt);
   520→    } catch (e) {
   521→      lastErr = e;
   522→      if (!isRetryableError(e) || attempt === MAX_RETRIES) {
   523→        if (attempt > 0) {
   524→          console.warn(`  [${label}] giving up after ${attempt + 1} attempt(s): ${(e?.shortMessage || e?.message || String(e)).slice(0, 100)}`);
   525→        }
   526→        throw e;
   527→      }
   528→      const delay = RETRY_BASE_MS * Math.pow(2, attempt) + Math.floor(Math.random() * 200);
   529→      console.warn(`  [${label}] attempt ${attempt + 1} failed (${(e?.shortMessage || e?.message || String(e)).slice(0, 80)}); retrying in ${delay}ms`);
   530→      await new Promise(r => setTimeout(r, delay));
   531→    }
   532→  }
   533→  throw lastErr;
   534→}

   537→async function pollOracle(entity) {
   538→  const url = `${ORACLE_API_URL}/api/v1/signal/${encodeURIComponent(entity)}`;
   539→  return withRetry(`poll:${entity.slice(0, 12)}`, () =>
   540→    axios.get(url, { timeout: 45000 }).then(r => r.data)
   541→  );
   542→}

   544→// ── Reflexive self-halt ────────────────────────────────────────────────────
   545→// Before publishing anything, check TRION's own self-verification status.
   546→// If TRION's own coherence has dropped below the SILENCE threshold, the
   547→// protocol does not trust its own signal generation right now — so the
   548→// relayer must not broadcast anything on-chain this cycle.
   549→async function checkSelfHalt() {
   550→  try {
   551→    const r = await axios.get(`${ORACLE_API_URL}/api/v1/self`, { timeout: 8000 });
   552→    const status = r.data?.status;
   553→    if (status === "SILENCED") {
   554→      console.warn(`[SELF-HALT] TRION self-coherence=${r.data.coherence} < threshold=${r.data.threshold} (limiting=${r.data.limiting_plane}) — skipping this cycle, no signals will be published`);
   555→      return true;
   556→    }
   557→    return false;
   558→  } catch (e) {
   559→    // If self-status is unreachable the relayer cannot confirm its own oracle
   560→    // is healthy — HALT this cycle (fail-closed).  Publishing signals from an
   561→    // oracle whose health is unknown risks propagating incorrect data on-chain.
   562→    console.warn(`[SELF-HALT] could not reach /api/v1/self (${e.message}) — HALTING this cycle (fail-closed)`);
   563→    return true;
   564→  }
   565→}

   569→  console.log(`\n[${stamp}] tick — entities=${MONITORED.length}, active_chains=${activeChains.length}, mode=${DRY_RUN ? "DRY_RUN" : "LIVE"}`);
