    28→# ── In-process rate limiter (sliding window, per-IP) ─────────────────────────
    29→# Phase 2.3: Hardened with background cleanup thread + env-configurable limits.
    30→#
    31→# NOTE: This is per-process. With gunicorn -w N, effective limit = N × _RL_MAX_REQS.
    32→# For production multi-instance deployments, replace with Redis-backed limiting
    33→# (e.g. flask-limiter + redis-py) so all workers share a single counter per IP.
    34→#
    35→# Defaults: 300 requests / 60 seconds per IP (production-grade for read-heavy API).
    36→# Override via env:
    37→#   RATE_LIMIT_WINDOW_SEC=60      — sliding window length in seconds
    38→#   RATE_LIMIT_MAX_REQUESTS=300   — max requests per window per IP
    39→#   RATE_LIMIT_CLEANUP_INTERVAL=300 — background cleanup sweep interval (seconds)
    40→#
    41→# Exempts /health and the static dashboard assets to allow monitoring probes.
    42→_RL_WINDOW          = int(os.environ.get("RATE_LIMIT_WINDOW_SEC", 60))
    43→_RL_MAX_REQS        = int(os.environ.get("RATE_LIMIT_MAX_REQUESTS", 300))
    44→_RL_CLEANUP_INTERVAL = int(os.environ.get("RATE_LIMIT_CLEANUP_INTERVAL", 300))
    45→_RL_STALE_AFTER      = 300  # IPs not seen in 5 minutes are evictable
    46→_RL_BUCKET_CAP       = 10_000  # hard cap on tracked IPs (prevents memory blowup)
    47→_rl_lock     = threading.Lock()
    48→_rl_buckets: dict = {}   # ip → deque of request timestamps
    49→
    50→def _get_client_ip() -> str:
    51→    """Return the real client IP, respecting X-Forwarded-For if behind a proxy."""
    52→    xff = request.headers.get("X-Forwarded-For", "")
    53→    if xff:
    54→        return xff.split(",")[0].strip()

    58→def _rl_cleanup_loop():
    59→    """Background thread that periodically evicts stale IP buckets.
    60→
    61→    Runs forever; checks every _RL_CLEANUP_INTERVAL seconds. This prevents
    62→    memory growth from abandoned IP buckets when traffic rotates through
    63→    many clients (e.g. CDN edges, mobile networks).
    64→    """
    65→    while True:
    66→        try:
    67→            time.sleep(_RL_CLEANUP_INTERVAL)
    68→            now = time.time()
    69→            stale_cutoff = now - _RL_STALE_AFTER
    70→            evicted = 0
    71→            with _rl_lock:
    72→                stale_keys = [
    73→                    k for k, v in _rl_buckets.items()
    74→                    if not v or v[-1] < stale_cutoff
    75→                ]
    76→                # Cap evictions per sweep to avoid holding the lock too long
    77→                for k in stale_keys[:2000]:
    78→                    del _rl_buckets[k]
    79→                    evicted += 1
    80→            if evicted:
    81→                _log.info("rate-limiter cleanup: evicted %d stale IP buckets", evicted)
    82→        except Exception as e:
    83→            _log.warning("rate-limiter cleanup error: %s", e)
    84→
    85→
    86→# Start the cleanup thread once at module load (daemon so it dies with the process)
    87→_rl_cleanup_thread = threading.Thread(
    88→    target=_rl_cleanup_loop, name="rl-cleanup", daemon=True
    89→)
    90→_rl_cleanup_thread.start()

    93→@app.before_request
    94→def _rate_limit():
    95→    # Allow health probes and static files through without counting
    96→    path = request.path
    97→    if path in ("/api/v1/health", "/favicon.ico") or path.startswith("/static/"):
    98→        return None
    99→
   100→    ip  = _get_client_ip()
   101→    now = time.time()
   102→
   103→    with _rl_lock:
   104→        bucket = _rl_buckets.setdefault(ip, deque())
   105→        # Evict timestamps outside the current window
   106→        cutoff = now - _RL_WINDOW
   107→        while bucket and bucket[0] < cutoff:
   108→            bucket.popleft()
   109→
   110→        if len(bucket) >= _RL_MAX_REQS:
   111→            retry_after = int(_RL_WINDOW - (now - bucket[0])) + 1
   112→            _log.warning("Rate limit exceeded: ip=%s requests=%d window=%ds", ip, len(bucket), _RL_WINDOW)
   113→            from flask import Response
   114→            return Response(
   115→                json.dumps({"error": "rate_limit_exceeded",
   116→                            "message": f"Too many requests — limit is {_RL_MAX_REQS} per {_RL_WINDOW}s",
   117→                            "retry_after_seconds": retry_after}),
   118→                status=429,
   119→                mimetype="application/json",
   120→                headers={"Retry-After": str(retry_after)},
   121→            )
   122→
   123→        bucket.append(now)
   124→
   125→        # Hard cap: if too many unique IPs tracked, evict oldest ones synchronously
   126→        # (fallback to the background thread for normal operation)
   127→        if len(_rl_buckets) > _RL_BUCKET_CAP:
   128→            stale_cutoff = now - _RL_STALE_AFTER
   129→            stale_keys = [k for k, v in _rl_buckets.items() if not v or v[-1] < stale_cutoff]
   130→            for k in stale_keys[:1000]:
   131→                del _rl_buckets[k]
   132→
   133→    return None

   463→# ── Behavioral plane computation — hash-seeded + live FAISS enrichment ───────
   464→# Hash gives stable deterministic base values per entity.
   465→# FAISS enrichment overwrites Mental and ANIMA with live indexed data
   466→# when real vectors exist for the entity (non-neutral-prior).
   467→
   468→def _entity_seed(eid: str) -> float:
   469→    # SHA3-256 — matches canonical BH L0.1 spec, Rust trion_common::bh_id(),
   470→    # and TypeScript entityIdFromAddr() in chains/shared/canonical_bh.ts.
   471→    # Using SHA-256 here was a bug: it broke cross-language BEO identity consistency.
   472→    h = hashlib.sha3_256(eid.encode()).digest()
   473→    return int.from_bytes(h[:4], "big") / 0xFFFFFFFF
   474→
   475→# Minimum coherence threshold used in COLD_START SILENCE signals ────────────
   476→# Mirrors THETA_MIN from src/core/coherence_engine.py (0.55) so COLD_START
   477→# responses carry a meaningful threshold field without importing the engine.
   478→THETA_MIN_COLD_START: float = 0.55
   479→
   480→# FAISS enrichment cache  ──────────────────────────────────────────────────
   481→# Phase 2.2: Bounded LRU cache with TTL + thread safety.
   482→# Replaces the unbounded dict that could grow without limit under sustained
   483→# traffic. Uses functools.lru_cache with a time-bucketed key to force refresh

   708→def _faiss_plane_timestamp(eid: str) -> float | None:
   709→    """Return the timestamp of the last FAISS cache entry for *eid*, or None."""
   710→    # Query each TTL bucket — the most recent non-None one is the live entry.
   711→    # We probe the current bucket and the previous one to cover the TTL boundary.

   715→        cached = _query_faiss_planes_cached(eid, bucket)
   716→        if cached is not None:
   717→            _result, ts = cached
   718→            return ts
   719→    return None

  1044→def _sanitize_string(s):
  1045→    if not isinstance(s, str):
  1046→        return s
  1047→    result = s
  1048→    for greek, eng in _GREEK_MAP.items():
  1049→        result = result.replace(greek, eng)
  1050→    for pattern, replacement in _FORMULA_REPLACEMENTS:
  1051→        result = _sanitize_re.sub(pattern, replacement, result)
  1052→    # Remove mathematical operators used in formulas
  1053→    result = _sanitize_re.sub(r'[·×÷±√∫∞≠≤≥→⇒⇔∈∉⊂⊃∪∩]', ' ', result)
  1054→    result = _sanitize_re.sub(r'\s+', ' ', result).strip()
  1055→    return result
  1056→
  1057→def _sanitize_obj(obj):
  1058→    if isinstance(obj, str):
  1059→        return _sanitize_string(obj)
  1060→    elif isinstance(obj, dict):
  1061→        return {k: _sanitize_obj(v) for k, v in obj.items()}
  1062→    elif isinstance(obj, list):
  1063→        return [_sanitize_obj(item) for item in obj]
  1064→    elif isinstance(obj, tuple):
  1065→        return tuple(_sanitize_obj(item) for item in obj)
  1066→    return obj

  1076→    except Exception:

  1078→    return response
