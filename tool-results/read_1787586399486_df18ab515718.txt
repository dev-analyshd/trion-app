   324→
   325→def _apply_cred_decay(source_id: str):
   326→    """
   327→    L3.4: CRED(s,t) = CRED(s,t-1) × 0.99_per_day.
   328→    Applies lazy compound decay since the last recorded decay timestamp.
   329→    """
   330→    now_ts = datetime.now(timezone.utc).timestamp()
   331→    with _db_lock:
   332→        conn = _db_conn()
   333→        row = conn.execute(
   334→            "SELECT cred, last_decay FROM anima_sources WHERE source_id = ?",
   335→            (source_id,)
   336→        ).fetchone()
   337→        if not row:
   338→            conn.close()
   339→            return
   340→        cred       = row["cred"]
   341→        last_decay = row["last_decay"] or now_ts
   342→        days_elapsed = (now_ts - last_decay) / 86400.0
   343→        if days_elapsed >= 1.0:
   344→            days_full = int(days_elapsed)
   345→            new_cred  = max(0.05, cred * (CRED_DECAY_DAILY ** days_full))
   346→            conn.execute("""
   347→                UPDATE anima_sources
   348→                SET cred = ?, last_decay = ?, updated_at = ?
   349→                WHERE source_id = ?
   350→            """, (new_cred, now_ts, datetime.now(timezone.utc).isoformat(), source_id))
   351→            conn.commit()
   352→            _cred_cache[source_id] = new_cred
   353→        else:
   354→            _cred_cache[source_id] = cred
   355→        conn.close()

   358→def update_cred(source_id: str, event_type: str, entity_id: str = "", note: str = ""):
   359→    """
   360→    L3.4: Apply a credibility event to a source.
   361→    event_type: VERIFIED | FALSIFIED | MANIPULATION | CONFLICT
   362→    """
   363→    delta_map = {
   364→        "VERIFIED":    CRED_DELTA_VERIFIED,
   365→        "FALSIFIED":   CRED_DELTA_FALSIFIED,
   366→        "MANIPULATION": CRED_DELTA_MANIPULATION,
   367→        "CONFLICT":    CRED_DELTA_CONFLICT,
   368→    }
   369→    delta = delta_map.get(event_type, 0.0)
   370→    if delta == 0.0:
   371→        return
   372→    _apply_cred_decay(source_id)
   373→    now_ts = datetime.now(timezone.utc).timestamp()
   374→    with _db_lock:
   375→        conn = _db_conn()
   376→        row = conn.execute(
   377→            "SELECT cred, total_events, verified, falsified FROM anima_sources WHERE source_id = ?",
   378→            (source_id,)
   379→        ).fetchone()
   380→        if not row:
   381→            conn.close()
   382→            return
   383→        new_cred = max(0.05, min(1.0, row["cred"] + delta))
   384→        flagged  = 1 if new_cred < CRED_FLAG_THRESHOLD else 0
   385→        excluded = 1 if new_cred < CRED_EXCLUDE_THRESHOLD else 0
   386→        conn.execute("""
   387→            UPDATE anima_sources
   388→            SET cred = ?, total_events = total_events + 1,
   389→                verified  = verified  + ?,
   390→                falsified = falsified + ?,
   391→                flagged   = ?,
   392→                excluded  = ?,
   393→                updated_at = ?
   394→            WHERE source_id = ?
   395→        """, (new_cred,
   396→              1 if event_type == "VERIFIED"  else 0,
   397→              1 if event_type == "FALSIFIED" else 0,
   398→              flagged, excluded,
   399→              datetime.now(timezone.utc).isoformat(),
   400→              source_id))
   401→        conn.execute("""
   402→            INSERT INTO anima_cred_events
   403→                (source_id, event_type, delta, cred_after, entity_id, note, ts)
   404→            VALUES (?, ?, ?, ?, ?, ?, ?)
   405→        """, (source_id, event_type, delta, new_cred, entity_id, note, now_ts))
   406→        conn.commit()
   407→        conn.close()
   408→    _cred_cache[source_id] = new_cred
   409→    logger.info("[CRED] %s %s → %.3f (Δ%.1f)", source_id, event_type, new_cred, delta)
   410→

   412→def get_all_cred_status() -> List[Dict]:
   413→    """Return current CRED status for all sources."""
   414→    for sid in SOURCES:
   415→        _apply_cred_decay(sid)
   416→    with _db_lock:
   417→        conn = _db_conn()
   418→        rows = conn.execute("""
   419→            SELECT source_id, category, cred, total_events, verified,
   420→                   falsified, flagged, excluded, updated_at
   421→            FROM anima_sources ORDER BY cred DESC
   422→        """).fetchall()
   423→        conn.close()
   424→    return [dict(r) for r in rows]

   454→def _edgar_fetch_filing_text(hit_id: str, timeout: int = 5) -> str:
   455→    """
   456→    Fetch actual 8-K / 10-K body text from SEC EDGAR archives.
   457→
   458→    EDGAR hit `_id` format: "edgar/data/{cik}/{accession_nodash}.json"
   459→    We reconstruct the filing index URL and fetch the primary document.
   460→    Returns extracted plain text, empty string on failure.
   461→    """
