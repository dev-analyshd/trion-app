    40→    40→#
    41→    41→# Exempts /health and the static dashboard assets to allow monitoring probes.
    42→    42→_RL_WINDOW          = int(os.environ.get("RATE_LIMIT_WINDOW_SEC", 60))
    43→    43→_RL_MAX_REQS        = int(os.environ.get("RATE_LIMIT_MAX_REQUESTS", 300))
    44→    44→_RL_CLEANUP_INTERVAL = int(os.environ.get("RATE_LIMIT_CLEANUP_INTERVAL", 300))
    45→    45→_RL_STALE_AFTER      = 300  # IPs not seen in 5 minutes are evictable
    46→    46→_RL_BUCKET_CAP       = 10_000  # hard cap on tracked IPs (prevents memory blowup)
    47→    47→_rl_lock     = threading.Lock()
    48→    48→_rl_buckets: dict = {}   # ip → deque of request timestamps

    50→    50→def _get_client_ip() -> str:
    51→    51→    """Return the real client IP, respecting X-Forwarded-For if behind a proxy."""
    52→    52→    xff = request.headers.get("X-Forwarded-For", "")
    53→    53→    if xff:
    54→    54→        return xff.split(",")[0].strip()
    55→    55→    return request.remote_addr or "unknown"

    58→    58→def _rl_cleanup_loop():
    59→    59→    """Background thread that periodically evicts stale IP buckets.
    60→    60→
    61→    61→    Runs forever; checks every _RL_CLEANUP_INTERVAL seconds. This prevents
    62→    62→    memory growth from abandoned IP buckets when traffic rotates through
    63→    63→    many clients (e.g. CDN edges, mobile networks).
    64→    64→    """
    65→    65→    while True:
    66→    66→        try:
    67→    67→            time.sleep(_RL_CLEANUP_INTERVAL)
    68→    68→            now = time.time()
    69→    69→            stale_cutoff = now - _RL_STALE_AFTER
    70→    70→            evicted = 0
    71→    71→            with _rl_lock:
    72→    72→                stale_keys = [
    73→    73→                    k for k, v in _rl_buckets.items()
    74→    74→                    if not v or v[-1] < stale_cutoff
    75→    75→                ]
    76→    76→                # Cap evictions per sweep to avoid holding the lock too long
    77→    77→                for k in stale_keys[:2000]:
    78→    78→                    del _rl_buckets[k]
    79→    79→                    evicted += 1
    80→    80→            if evicted:
    81→    81→                _log.info("rate-limiter cleanup: evicted %d stale IP buckets", evicted)
    82→    82→        except Exception as e:
    83→    83→            _log.warning("rate-limiter cleanup error: %s", e)
    84→    84→
    85→    85→
    86→    86→# Start the cleanup thread once at module load (daemon so it dies with the process)
    87→    87→_rl_cleanup_thread = threading.Thread(
    88→    88→    target=_rl_cleanup_loop, name="rl-cleanup", daemon=True
    89→    89→)
    90→    90→_rl_cleanup_thread.start()

    93→    93→@app.before_request
    94→    94→def _rate_limit():
    95→    95→    # Allow health probes and static files through without counting
    96→    96→    path = request.path
    97→    97→    if path in ("/api/v1/health", "/favicon.ico") or path.startswith("/static/"):
    98→    98→        return None
    99→    99→
   100→   100→    ip  = _get_client_ip()
   101→   101→    now = time.time()
   102→   102→
   103→   103→    with _rl_lock:
   104→   104→        bucket = _rl_buckets.setdefault(ip, deque())
   105→   105→        # Evict timestamps outside the current window
   106→   106→        cutoff = now - _RL_WINDOW
   107→   107→        while bucket and bucket[0] < cutoff:
   108→   108→            bucket.popleft()
   109→   109→
   110→   110→        if len(bucket) >= _RL_MAX_REQS:
   111→   111→            retry_after = int(_RL_WINDOW - (now - bucket[0])) + 1
   112→   112→            _log.warning("Rate limit exceeded: ip=%s requests=%d window=%ds", ip, len(bucket), _RL_WINDOW)
   113→   113→            from flask import Response
   114→   114→            return Response(
   115→   115→                json.dumps({"error": "rate_limit_exceeded",
   116→   116→                            "message": f"Too many requests — limit is {_RL_MAX_REQS} per {_RL_WINDOW}s",
   117→   117→                            "retry_after_seconds": retry_after}),
   118→   118→                status=429,
   119→   119→                mimetype="application/json",
   120→   120→                headers={"Retry-After": str(retry_after)},
   121→   121→            )
   122→   122→
   123→   123→        bucket.append(now)
   124→   124→
   125→   125→        # Hard cap: if too many unique IPs tracked, evict oldest ones synchronously
   126→   126→        # (fallback to the background thread for normal operation)
   127→   127→        if len(_rl_buckets) > _RL_BUCKET_CAP:
   128→   128→            stale_cutoff = now - _RL_STALE_AFTER
   129→   129→            stale_keys = [k for k, v in _rl_buckets.items() if not v or v[-1] < stale_cutoff]
   130→   130→            for k in stale_keys[:1000]:
   131→   131→                del _rl_buckets[k]
   132→   132→
   133→   133→    return None
