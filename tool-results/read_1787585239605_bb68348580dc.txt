   397→function Home() {
   398→  const searchParams = useSearchParams();
   399→  const router = useRouter();
   400→  const initialPage = searchParams.get('page') || 'dashboard';
   401→  const [activePage, setActivePage] = useState(
   402→    PAGE_MAP[initialPage] ? initialPage : 'dashboard'
   403→  );
   404→  const [sidebarOpen, setSidebarOpen] = useState(false);
   405→  const [settingsOpen, setSettingsOpen] = useState(false);
   406→  const [theme, toggleTheme] = useTheme();
   407→
   408→  const changePage = useCallback((p: string) => {
   409→    setActivePage(p);
   410→    const url = new URL(window.location.href);
   411→    url.searchParams.set('page', p);
   412→    router.push(`${url.pathname}?${url.searchParams.toString()}`, { scroll: false });
   413→  }, [router]);
   414→
   415→  useEffect(() => {
   416→    const p = searchParams.get('page');
   417→    if (p && PAGE_MAP[p] && p !== activePage) {
   418→      setActivePage(p);
   419→    }
   420→  }, [searchParams, activePage]);
   421→
   422→  // Keyboard shortcuts
   423→  useEffect(() => {
   424→    const onKey = (e: KeyboardEvent) => {
   425→      const t = e.target as HTMLElement;
   426→      if (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable) return;
   427→      if (e.key === '?' || (e.shiftKey && e.key === '/')) {
   428→        e.preventDefault();
   429→        alert(
   430→          'TRION Keyboard Shortcuts\n\n' +
   431→          '⌘K / Ctrl+K    Command palette\n' +
   432→          '⌘B / Ctrl+B    Toggle sidebar\n' +
   433→          '?                This help\n' +
   434→          'Esc             Close overlays\n'
   435→        );
   436→      }
   437→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'b') {
   438→        e.preventDefault();
   439→        setSidebarOpen(v => !v);
   440→      }
   441→    };
   442→    window.addEventListener('keydown', onKey);
   443→    return () => window.removeEventListener('keydown', onKey);
   444→  }, []);
   445→
   446→  const PageComponent = PAGE_MAP[activePage] || RedesignedDashboard;
   447→  const pageTitle = PAGE_TITLES[activePage] || 'Dashboard';
   448→
   449→  const palettePages = useMemo(
   450→    () => NAV.flatMap(g => g.items.map(item => ({
   451→      id: item.id,
   452→      label: item.label,
   453→      group: g.label,
   454→    }))),
   455→    [],
   456→  );
   457→
   458→  return (
   459→    <div className="flex h-screen overflow-hidden bg-background">
   460→      <CommandPalette pages={palettePages} onSelect={changePage} />
   461→      <Sidebar
   462→        activePage={activePage}
   463→        onPageChange={changePage}
   464→        isOpen={sidebarOpen}
   465→        onClose={() => setSidebarOpen(false)}
   466→      />
   467→      <div className="flex-1 flex flex-col overflow-hidden">
   468→        {/* Top bar */}
   469→        <header className="flex items-center justify-between px-4 md:px-6 h-16 border-b border-border bg-card flex-shrink-0 gap-2">
   470→          <div className="flex items-center gap-3 min-w-0 flex-1">
   471→            <button
   472→              onClick={() => setSidebarOpen(true)}
   473→              className="lg:hidden p-2 rounded-lg hover:bg-accent flex-shrink-0"
   474→              aria-label="Open sidebar"
   475→            >
   476→              <Icons.Menu className="w-5 h-5" />
   477→            </button>
   478→            <div className="min-w-0">
   479→              <h1 className="text-lg font-bold truncate">{pageTitle}</h1>
   480→              <p className="text-xs text-muted-foreground hidden sm:block truncate">
   481→                TRION Protocol — Behavioral Truth Oracle
   482→              </p>
   483→            </div>
   484→          </div>
   485→          <div className="flex items-center gap-2">
   486→            <LiveClock />
   487→            <button
   488→              onClick={toggleTheme}
   489→              className="p-2 rounded-lg hover:bg-accent"
   490→              aria-label="Toggle theme"
   491→            >
   492→              {theme === 'dark' ? <Icons.Sun className="w-4 h-4" /> : <Icons.Moon className="w-4 h-4" />}
   493→            </button>
   494→            <button
   495→              onClick={() => setSettingsOpen(true)}
   496→              className="p-2 rounded-lg hover:bg-accent"
   497→              aria-label="Settings"
   498→            >
   499→              <Icons.Settings className="w-4 h-4" />
   500→            </button>
   501→            <WalletButton />
   502→          </div>
   503→        </header>
   504→
   505→        {/* Main content */}
   506→        <main className="flex-1 overflow-y-auto p-4 md:p-6">
   507→          <PageComponent />
   508→        </main>
   509→      </div>
   510→      <SettingsModal open={settingsOpen} onClose={() => setSettingsOpen(false)} />
   511→    </div>
   512→  );
   513→}

   700→function MetricCard({ icon, label, value, sub, color }: {
   701→  icon: React.ReactNode;
   702→  label: string;
   703→  value: string;
   704→  sub?: string;
   705→  color: 'blue' | 'purple' | 'green' | 'amber';
   706→}) {
   707→  const colors = {
   708→    blue: 'from-blue-500/20 to-blue-500/5 text-blue-400 border-blue-500/20',
   709→    purple: 'from-purple-500/20 to-purple-500/5 text-purple-400 border-purple-500/20',
   710→    green: 'from-green-500/20 to-green-500/5 text-green-400 border-green-500/20',
   711→    amber: 'from-amber-500/20 to-amber-500/5 text-amber-400 border-amber-500/20',
   712→  };
   713→  return (
   714→    <div className={`bg-gradient-to-br ${colors[color]} border rounded-xl p-4`}>
   715→      <div className="flex items-center gap-2 mb-2 opacity-80">
   716→        {icon}
   717→        <span className="text-xs font-medium uppercase tracking-wider">{label}</span>
   718→      </div>
   719→      <div className="font-mono text-2xl font-bold">{value}</div>
   720→      {sub && <div className="text-xs opacity-70 mt-1">{sub}</div>}
   721→    </div>
   722→  );
   723→}
