   107→-- ── L2.1 Akashic Depth View ───────────────────────────────────────────────────
   108→-- D(t) = accumulated depth of truth. Appended to every TRIONSignal.
   109→CREATE OR REPLACE VIEW akashic_depth AS
   110→SELECT
   111→    entity_id,
   112→    COUNT(*)                                               AS record_count,
   113→    SUM(entropy_delta)                                     AS total_entropy,
   114→    SUM(magnitude_norm * entropy_delta)                    AS raw_depth,
   115→    MIN(time)                                              AS genesis_time,
   116→    MAX(time)                                              AS last_seen,
   117→    EXTRACT(EPOCH FROM (MAX(time) - MIN(time)))            AS lifespan_seconds,
   118→    COUNT(*) / GREATEST(
   119→        EXTRACT(EPOCH FROM (MAX(time) - MIN(time))) / 86400.0, 1
   120→    )                                                      AS daily_activity_rate
   121→FROM akashic_bh
   122→GROUP BY entity_id;

   667→-- Append-only enforcement: sanctions records cannot be deleted (only delisted)
   668→CREATE OR REPLACE FUNCTION prevent_sanctions_delete()
   669→RETURNS TRIGGER AS $$
   670→BEGIN
   671→    RAISE EXCEPTION 'Sanctions registry is append-only (AWA-protected). Use deactivate instead of DELETE.';
   672→END;
   673→$$ LANGUAGE plpgsql;

   675→DROP TRIGGER IF EXISTS sanctions_append_only ON sanctions_registry;
   676→CREATE TRIGGER sanctions_append_only
   677→BEFORE DELETE ON sanctions_registry
   678→FOR EACH ROW EXECUTE FUNCTION prevent_sanctions_delete();
