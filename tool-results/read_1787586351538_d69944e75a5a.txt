   225→def _seed(chain_id: str) -> int:
   226→    """Deterministic 32-bit seed from chain id string."""
   227→    h = hashlib.sha256(chain_id.encode()).digest()
   228→    return int.from_bytes(h[:4], "big")

   231→def get_bh_stats(chain: dict, live_records: int | None = None) -> dict:
   232→    """
   233→    Return BH stats for a chain.
   234→
   235→    HONEST DATA MODEL (audit remediation):
   236→      - When `live_records` (real bh_ledger.db count for this chain) is
   237→        available, bh_proofs is the REAL count and stats_source="ledger".
   238→      - Otherwise the deterministic baseline is returned explicitly labeled
   239→        stats_source="estimated" — capacity-planning figures derived from
   240→        the chain's status/VM, NOT live measurements. Never presented as
   241→        indexed data.
   242→    """
   243→    s = _seed(chain["id"])
   244→    status = chain["status"]
   245→    vm = chain["vm"]
   246→
   247→    if live_records is not None:
   248→        return {
   249→            "bh_proofs": int(live_records),
   250→            "faiss_vectors": int(live_records),   # every BH has a FAISS vector
   251→            "last_block": 0,
   252→            "last_indexed_ts": None,
   253→            "indexer": chain.get("indexer", "trion-evm"),
   254→            "stats_source": "ledger",
   255→        }
   256→
   257→    # Deterministic capacity estimate (NOT live data)
   258→    base_proofs = _PROOF_BASE.get(status, 45_000)
   259→    base_block = _BLOCK_BASE.get(vm, 10_000_000)
   260→
   261→    # jitter within ±30% using seed
   262→    jitter = (s % 1000) / 1000.0 * 0.6 - 0.3
   263→    bh_proofs = int(base_proofs * (1 + jitter))
   264→    last_block = int(base_block * (1 + jitter * 0.5))
   265→
   266→    # FAISS vectors: 70-90% of proof count
   267→    faiss_frac = 0.70 + (s % 200) / 1000.0
   268→    faiss_vectors = int(bh_proofs * faiss_frac)
   269→
   270→    return {
   271→        "bh_proofs": bh_proofs,
   272→        "faiss_vectors": faiss_vectors,
   273→        "last_block": last_block,
   274→        "last_indexed_ts": None,   # never fabricate freshness for estimates
   275→        "indexer": chain.get("indexer", "trion-evm"),
   276→        "stats_source": "estimated",
   277→    }

   280→def enrich(chain: dict, live_records: int | None = None) -> dict:
   281→    """Return chain dict enriched with BH stats (real ledger counts preferred)."""
   282→    return {**chain, **get_bh_stats(chain, live_records)}

   285→def get_live_chain_stats() -> dict:
   286→    """
   287→    Query bh_ledger.db for real per-chain BH record counts.
   288→    Returns {chain_label: count} from live SQLite data.
   289→    Falls back to empty dict if the DB is unavailable.
   290→    Result is cached for 60 s to avoid per-request DB hits.
   291→    """
   292→    import sqlite3 as _sql
   293→    import os as _os
   294→    import time as _t
   295→    import threading as _th
   296→    _f = get_live_chain_stats
   297→    if not hasattr(_f, "_cache"):
   298→        _f._cache = {}
   299→        _f._ts    = 0.0
   300→        _f._lock  = _th.Lock()
   301→    now = _t.time()
   302→    with _f._lock:
   303→        if _f._cache and (now - _f._ts) < 60:
   304→            return dict(_f._cache)
   305→    db_path = _os.path.normpath(
   306→        _os.path.join(_os.path.dirname(__file__), "..", "bh_ledger.db")
   307→    )
   308→    try:
   309→        conn = _sql.connect(db_path, timeout=3.0)
   310→        conn.execute("PRAGMA query_only=1")
   311→        rows = conn.execute(
   312→            "SELECT chain_label, COUNT(*) FROM bh_ledger GROUP BY chain_label"
   313→        ).fetchall()
   314→        conn.close()
   315→        result = {r[0]: r[1] for r in rows}
   316→    except Exception:
   317→        result = {}
   318→    with _f._lock:
   319→        _f._cache = result
   320→        _f._ts    = now
   321→    return result

   328→def get_enriched_chains():
   329→    """Return all chains enriched with BH stats — REAL ledger counts preferred.
   330→
   331→    Audit remediation: chains with live bh_ledger.db records report the actual
   332→    count with stats_source="ledger"; all others are explicitly labeled
   333→    stats_source="estimated" (deterministic capacity figures, never presented
   334→    as indexed state).
   335→    """
   336→    live_stats = get_live_chain_stats()
   337→    enriched = []
   338→    for c in CHAINS:
   339→        # Resolve the bh_ledger chain_label for this catalog entry
   340→        explicit = c.get("bh_label")
   341→        if explicit:
   342→            label_candidates = [explicit]
   343→        else:
   344→            label_candidates = [
   345→                c["id"].upper().replace("-", "_"),
   346→                c["name"].upper().replace(" ", "_").replace("-", "_"),
   347→                c["id"].upper().replace("-", "_") + "_MAINNET",
   348→                c["id"].upper().replace("-", "_") + "_DEVNET",
   349→            ]
   350→        live_records = None
   351→        for lbl in label_candidates:
   352→            if lbl in live_stats:
   353→                live_records = live_stats[lbl]
   354→                break
   355→
   356→        e = enrich(c, live_records)
   357→        if live_records is not None:
   358→            e["bh_records_live"] = live_records
   359→        enriched.append(e)
   360→    return enriched
